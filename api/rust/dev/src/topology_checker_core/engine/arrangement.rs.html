20// Original vertices stay compact. Rational values are allocated only for actual
21// intersections and face witnesses, rather than for every input coordinate.
22#[derive(Debug, Clone)]
23pub(super) enum Exact {
24    Input(Coord),
25    Rational(Box<RationalNode>),
26}
27#[derive(Debug, Clone)]
28pub(super) struct RationalNode {
29    xy: (Q, Q),
30    rounded: Coord,
31}
32impl Exact {
33    fn input(p: Coord) -> Self {
34        Self::Input(p)
35    }
36    pub(super) fn rational(x: Q, y: Q) -> Self {
37        let rounded = Coord {
38            x: x.to_f64().expect("finite intersection"),
39            y: y.to_f64().expect("finite intersection"),
40        };
41        Self::Rational(Box::new(RationalNode {
42            xy: (x, y),
43            rounded,
44        }))
45    }
46    pub(super) fn xy(&self) -> (Q, Q) {
47        match self {
48            Self::Input(p) => (
49                Q::from_float(p.x).expect("finite input"),
50                Q::from_float(p.y).expect("finite input"),
51            ),
52            Self::Rational(p) => p.xy.clone(),
53        }
54    }
55    pub(super) fn output(&self) -> Coord {
56        match self {
57            Self::Input(p) => *p,
58            Self::Rational(p) => p.rounded,
59        }
60    }
61    fn compare(&self, other: &Self) -> Ordering {
62        certified::compare(self, other, 0).then_with(|| certified::compare(self, other, 1))
63    }
64}
65pub(super) fn orient(a: &Exact, b: &Exact, c: &Exact) -> Q {
66    if let (Exact::Input(a), Exact::Input(b), Exact::Input(c)) = (a, b, c) {
67        return Q::from_integer(match RobustKernel::orient2d(*a, *b, *c) {
68            Orientation::Clockwise => (-1).into(),
69            Orientation::Collinear => 0.into(),
70            Orientation::CounterClockwise => 1.into(),
71        });
72    }
73    let [ax, ay] = certified::coordinates(a);
74    let [bx, by] = certified::coordinates(b);
75    let [cx, cy] = certified::coordinates(c);
76    if let Some(order) = bx
77        .sub(ax)
78        .mul(cy.sub(ay))
79        .sub(by.sub(ay).mul(cx.sub(ax)))
80        .sign()
81    {
82        return Q::from_integer(match order {
83            Ordering::Less => (-1).into(),
84            Ordering::Greater => 1.into(),
85            Ordering::Equal => 0.into(),
86        });
87    }
88    let (ax, ay) = a.xy();
89    let (bx, by) = b.xy();
90    let (cx, cy) = c.xy();
91    (bx - &ax) * (cy - &ay) - (by - ay) * (cx - ax)
92}
93pub(super) fn on(a: &Exact, b: &Exact, p: &Exact) -> bool {
94    if !orient(a, b, p).is_zero() {
95        return false;
96    }
97    (0..2).all(|axis| {
98        let x = certified::compare(p, a, axis);
99        let y = certified::compare(p, b, axis);
100        x.is_eq() || y.is_eq() || x != y
101    })
102}
103pub(super) fn intersects(a: &Exact, b: &Exact, c: &Exact, d: &Exact) -> bool {
104    let ab_c = orient(a, b, c);
105    let ab_d = orient(a, b, d);
106    let cd_a = orient(c, d, a);
107    let cd_b = orient(c, d, b);
108    (ab_c.is_zero() && on(a, b, c))
109        || (ab_d.is_zero() && on(a, b, d))
110        || (cd_a.is_zero() && on(c, d, a))
111        || (cd_b.is_zero() && on(c, d, b))
112        || ab_c.cmp(&Q::zero()) != ab_d.cmp(&Q::zero())
113            && !ab_c.is_zero()
114            && !ab_d.is_zero()
115            && cd_a.cmp(&Q::zero()) != cd_b.cmp(&Q::zero())
116            && !cd_a.is_zero()
117            && !cd_b.is_zero()
118}
119fn crossing(a: Line, b: Line) -> Exact {
120    let (_, [x, y]) = super::line_graph::exact_crossing(a, b).expect("proper crossing");
121    Exact::rational(x, y)
122}

138// All rationals here are normalized by num-rational's constructors and arithmetic.
139// Hashing canonical numerators and denominators avoids Ratio's Euclidean hash,
140// which supports unnormalized external values and repeatedly divides big integers.
141struct RationalKey((Q, Q));
142impl PartialEq for RationalKey {
143    fn eq(&self, b: &Self) -> bool {
144        self.0.0.numer() == b.0.0.numer()
145            && self.0.0.denom() == b.0.0.denom()
146            && self.0.1.numer() == b.0.1.numer()
147            && self.0.1.denom() == b.0.1.denom()
148    }
149}
150impl Eq for RationalKey {}
151impl Hash for RationalKey {
152    fn hash<H: Hasher>(&self, state: &mut H) {
153        self.0.0.numer().hash(state);
154        self.0.0.denom().hash(state);
155        self.0.1.numer().hash(state);
156        self.0.1.denom().hash(state);
157    }
158}
159fn node(point: Exact, nodes: &mut Vec<Exact>, ids: &mut NodeIds) -> usize {
160    if let Exact::Rational(p) = &point {
161        let rounded = point.output();
162        // Only original vertices are fixed during output rounding. A binary
163        // representable intersection still belongs to the local solver.
164        if let Some(&id) = ids.inputs.get(&(
165            super::cluster::bits(rounded.x),
166            super::cluster::bits(rounded.y),
167        )) && Exact::input(rounded).xy() == p.xy
168        {
169            return id;
170        }
171    }
172    let id = nodes.len();
173    let old = match &point {
174        Exact::Input(p) => ids
175            .inputs
176            .insert((super::cluster::bits(p.x), super::cluster::bits(p.y)), id),
177        Exact::Rational(p) => ids.rational.insert(RationalKey(p.xy.clone()), id),
178    };
179    if let Some(old) = old {
180        // Restore the existing identity; a contact may occur on many segments.
181        match &point {
182            Exact::Input(p) => {
183                ids.inputs
184                    .insert((super::cluster::bits(p.x), super::cluster::bits(p.y)), old);
185            }
186            Exact::Rational(p) => {
187                ids.rational.insert(RationalKey(p.xy.clone()), old);
188            }
189        }
190        old
191    } else {
192        nodes.push(point);
193        id
194    }
195}
196fn ends(edge: usize, edges: &[Edge]) -> [usize; 2] {
197    let nodes = edges[edge / 2].nodes;
198    if edge.is_multiple_of(2) {
199        nodes
200    } else {
201        [nodes[1], nodes[0]]
202    }
203}
204pub(super) fn angle(a: &Exact, b: &Exact, origin: &Exact) -> Ordering {
205    if let (Exact::Input(a), Exact::Input(b), Exact::Input(o)) = (a, b, origin) {
206        let half = |p: &Coord| p.y < o.y || p.y == o.y && p.x < o.x;
207        return half(a)
208            .cmp(&half(b))
209            .then_with(|| match RobustKernel::orient2d(*o, *a, *b) {
210                Orientation::Clockwise => Ordering::Greater,
211                Orientation::CounterClockwise => Ordering::Less,
212                Orientation::Collinear => Ordering::Equal,
213            });
214    }
215    let half = |p: &Exact| {
216        let y = certified::compare(p, origin, 1);
217        y.is_lt() || y.is_eq() && certified::compare(p, origin, 0).is_lt()
218    };
219    half(a)
220        .cmp(&half(b))
221        .then_with(|| orient(origin, a, b).cmp(&Q::zero()).reverse())
222}
223fn envelope(a: Coord, b: Coord) -> rstar::AABB<[f64; 2]> {
224    rstar::AABB::from_corners(
225        [a.x.min(b.x).next_down(), a.y.min(b.y).next_down()],
226        [a.x.max(b.x).next_up(), a.y.max(b.y).next_up()],
227    )
228}
229/// Choose a rational point in the left face, with an exact clearance check.
230fn witness(h: usize, edges: &[Edge], nodes: &[Exact], tree: &index::Index) -> Exact {
231    let [a, b] = ends(h, edges);
232    let (a, b) = (&nodes[a], &nodes[b]);
233    let two = Q::from_integer(2.into());
234    let (ax, ay) = a.xy();
235    let (bx, by) = b.xy();
236    let mx = (&ax + &bx) / &two;
237    let my = (&ay + &by) / &two;
238    let middle = Exact::rational(mx.clone(), my.clone());
239    let mut divisor = Q::from_integer(1_048_576.into());
240    for _ in 0..4096 {
241        let p = Exact::rational(&mx - (&by - &ay) / &divisor, &my + (&bx - &ax) / &divisor);
242        let clear = tree
243            .locate_in_envelope_intersecting(&envelope(middle.output(), p.output()))
244            .all(|item| {
245                if item.data == h / 2 {
246                    return true;
247                }
248                let [c, d] = edges[item.data].nodes;
249                !intersects(&middle, &p, &nodes[c], &nodes[d])
250            });
251        if clear {
252            return p;
253        }
254        divisor *= &two;
255    }
256    panic!("rational face witness failed to make progress");
257}
258fn contains_exact(location: &Location, point: &Exact) -> bool {
259    position_exact(location, point) == 1
260}
261pub(super) fn position_exact(location: &Location, point: &Exact) -> i8 {
262    let p = point.output();
263    let query = rstar::AABB::from_corners(
264        [p.x.next_down(), p.y.next_down()],
265        [f64::MAX, p.y.next_up()],
266    );
267    let mut inside = false;
268    for item in location.tree.locate_in_envelope_intersecting(&query) {
269        let s = location.segments[item.data];
270        let a = Exact::input(s.start);
271        let b = Exact::input(s.end);
272        if on(&a, &b, point) {
273            return 0;
274        }
275        if certified::compare(&a, point, 1).is_gt() != certified::compare(&b, point, 1).is_gt()
276            && (orient(&a, &b, point) > Q::zero()) == (s.end.y > s.start.y)
277        {
278            inside = !inside;
279        }
280    }
281    if inside { 1 } else { -1 }
282}
283fn selected(w: [i32; 2], intersection: bool) -> bool {
284    if intersection {
285        w[0] > 0 && w[1] > 0
286    } else {
287        w[0] + w[1] > 0
288    }
289}

291fn component(polygons: &[(&Polygon, usize)], intersection: bool) -> super::precision::Component {
292    let mut sources = Vec::new();
293    for &(p, operand) in polygons {
294        for (i, ring) in std::iter::once(p.exterior())
295            .chain(p.interiors())
296            .enumerate()
297        {
298            let ccw = ring.winding_order() == Some(WindingOrder::CounterClockwise);
299            let mut delta = [0, 0];
300            delta[operand] = if ccw == (i == 0) { 1 } else { -1 };
301            sources.extend(ring.lines().map(|line| Source { line, delta }));
302        }
303    }
304    let tree = index::build(
305        sources
306            .iter()
307            .enumerate()
308            .map(|(i, s)| (i, s.line.bounding_rect())),
309    );
310    let mut nodes = Vec::new();
311    let mut ids = NodeIds::default();
312    let mut cuts = sources
313        .iter()
314        .map(|s| {
315            vec![
316                node(Exact::input(s.line.start), &mut nodes, &mut ids),
317                node(Exact::input(s.line.end), &mut nodes, &mut ids),
318            ]
319        })
320        .collect::<Vec<_>>();
321    for (i, s) in sources.iter().enumerate() {
322        crate::cancel::check();
323        for item in tree.locate_in_envelope_intersecting(&index::envelope(s.line.bounding_rect())) {
324            let j = item.data;
325            if j <= i {
326                continue;
327            }
328            if let Some(contact) = line_intersection(s.line, sources[j].line) {
329                match contact {
330                    LineIntersection::SinglePoint {
331                        intersection,
332                        is_proper,
333                    } => {
334                        let p = if is_proper {
335                            crossing(s.line, sources[j].line)
336                        } else {
337                            Exact::input(intersection)
338                        };
339                        let id = node(p, &mut nodes, &mut ids);
340                        cuts[i].push(id);
341                        cuts[j].push(id);
342                    }
343                    LineIntersection::Collinear { intersection } => {
344                        for p in [intersection.start, intersection.end] {
345                            let id = node(Exact::input(p), &mut nodes, &mut ids);
346                            cuts[i].push(id);
347                            cuts[j].push(id);
348                        }
349                    }
350                }
351            }
352        }
353    }
354    drop(ids);
355    drop(tree);
356    let mut source_order = Vec::new();
357    let mut edges: Vec<Edge> = Vec::new();
358    let mut keys = HashMap::new();
359    for (s, mut cuts) in sources.iter().zip(cuts) {
360        crate::cancel::check();
361        cuts.sort_unstable_by(|&a, &b| nodes[a].compare(&nodes[b]));
362        cuts.dedup();
363        // Source direction must use the same geometric order as its cut nodes.
364        // IEEE signed zero denotes one coordinate, not two distinct x positions.
365        let forward =
366            Exact::input(s.line.start).compare(&Exact::input(s.line.end)) == Ordering::Less;
367        for pair in cuts.windows(2) {
368            let a = pair[0];
369            let b = pair[1];
370            let axis = usize::from(
371                (s.line.end.y - s.line.start.y).abs() > (s.line.end.x - s.line.start.x).abs(),
372            );
373            source_order.push((a, b, axis));
374            let key = (a.min(b), a.max(b));
375            let id = *keys.entry(key).or_insert_with(|| {
376                let id = edges.len();
377                edges.push(Edge {
378                    nodes: [key.0, key.1],
379                    delta: [0, 0],
380                });
381                id
382            });
383            let sign = if (a == key.0) == forward { 1 } else { -1 };
384            for k in 0..2 {
385                edges[id].delta[k] += sign * s.delta[k];
386            }
387        }
388    }
389    drop(keys);
390    drop(sources);
391    let mut outgoing = vec![Vec::new(); nodes.len()];
392    for h in 0..edges.len() * 2 {
393        outgoing[ends(h, &edges)[0]].push(h);
394    }
395    for (id, hs) in outgoing.iter_mut().enumerate() {
396        crate::cancel::check();
397        hs.sort_unstable_by(|&a, &b| {
398            angle(
399                &nodes[ends(a, &edges)[1]],
400                &nodes[ends(b, &edges)[1]],
401                &nodes[id],
402            )
403        });
404    }
405    let mut next = vec![0; edges.len() * 2];
406    for hs in &outgoing {
407        for (i, &h) in hs.iter().enumerate() {
408            next[h ^ 1] = hs[(i + hs.len() - 1) % hs.len()];
409        }
410    }
411    let mut faces = vec![usize::MAX; next.len()];
412    let mut representatives = Vec::new();
413    for start in 0..next.len() {
414        crate::cancel::check();
415        if faces[start] != usize::MAX {
416            continue;
417        }
418        let id = representatives.len();
419        representatives.push(start);
420        let mut h = start;
421        loop {
422            assert_eq!(faces[h], usize::MAX, "face traversal must progress");
423            faces[h] = id;
424            h = next[h];
425            if h == start {
426                break;
427            }
428        }
429    }
430    let tree = index::build(edges.iter().enumerate().map(|(i, e)| {
431        (
432            i,
433            Rect::new(nodes[e.nodes[0]].output(), nodes[e.nodes[1]].output()),
434        )
435    }));
436    let locations = polygons
437        .iter()
438        .map(|&(p, operand)| (Location::new(p), operand))
439        .collect::<Vec<_>>();
440    let location_bounds = index::build(
441        polygons
442            .iter()
443            .enumerate()
444            .filter_map(|(i, (p, _))| p.bounding_rect().map(|r| (i, r))),
445    );
446    let mut adjacent = vec![Vec::new(); representatives.len()];
447    for (i, e) in edges.iter().enumerate() {
448        let left = faces[i * 2];
449        let right = faces[i * 2 + 1];
450        adjacent[right].push((left, e.delta));
451        adjacent[left].push((right, e.delta.map(|d| -d)));
452    }
453    let mut winding = vec![None; representatives.len()];
454    for seed in 0..representatives.len() {
455        if winding[seed].is_some() {
456            continue;
457        }
458        let sample = witness(representatives[seed], &edges, &nodes, &tree);
459        let mut value = [0, 0];
460        let p = sample.output();
461        let query = rstar::AABB::from_corners(
462            [p.x.next_down(), p.y.next_down()],
463            [p.x.next_up(), p.y.next_up()],
464        );
465        for item in location_bounds.locate_in_envelope_intersecting(&query) {
466            let (location, operand) = &locations[item.data];
467            if contains_exact(location, &sample) {
468                value[*operand] += 1;
469            }
470        }
471        winding[seed] = Some(value);
472        let mut queue = VecDeque::from([seed]);
473        while let Some(face) = queue.pop_front() {
474            let value = winding[face].expect("queued face");
475            for &(other, delta) in &adjacent[face] {
476                let value = [value[0] + delta[0], value[1] + delta[1]];
477                if let Some(old) = winding[other] {
478                    assert_eq!(old, value, "consistent face winding");
479                } else {
480                    winding[other] = Some(value);
481                    queue.push_back(other);
482                }
483            }
484        }
485    }
486    drop(tree);
487    drop(locations);
488    drop(adjacent);
489    let keep = (0..next.len())
490        .map(|h| {
491            selected(winding[faces[h]].expect("classified face"), intersection)
492                && !selected(
493                    winding[faces[h ^ 1]].expect("classified face"),
494                    intersection,
495                )
496        })
497        .collect::<Vec<_>>();
498    let mut boundary_next = vec![usize::MAX; next.len()];
499    for h in 0..next.len() {
500        if !keep[h] {
501            continue;
502        }
503        let hs = &outgoing[ends(h, &edges)[1]];
504        let i = hs.iter().position(|&v| v == h ^ 1).expect("twin incidence");
505        boundary_next[h] = (1..=hs.len())
506            .map(|k| hs[(i + hs.len() - k) % hs.len()])
507            .find(|&v| keep[v])
508            .expect("boundary continuation");
509    }
510    let mut visited = vec![false; next.len()];
511    let mut shells = Vec::new();
512    let mut holes = Vec::new();
513    for start in 0..next.len() {
514        crate::cancel::check();
515        if !keep[start] || visited[start] {
516            continue;
517        }
518        let mut h = start;
519        let mut ring = Vec::new();
520        loop {
521            assert!(!visited[h], "boundary traversal must progress");
522            visited[h] = true;
523            ring.push(ends(h, &edges)[0]);
524            h = boundary_next[h];
525            if h == start {
526                break;
527            }
528        }
529        ring.push(ring[0]);
530        // Split by exact node identity, before any coordinate conversion.
531        let mut path = Vec::new();
532        let mut positions = HashMap::new();
533        for id in ring {
534            if let Some(&start) = positions.get(&id) {
535                let mut cycle = vec![id];
536                let tail = path.split_off(start + 1);
537                for p in &tail {
538                    positions.remove(p);
539                }
540                cycle.extend(tail);
541                cycle.push(id);
542                match super::precision::area_sign(&nodes, &cycle) {
543                    Ordering::Greater => shells.push(cycle),
544                    Ordering::Less => holes.push(cycle),
545                    Ordering::Equal => {}
546                }
547            } else {
548                positions.insert(id, path.len());
549                path.push(id);
550            }
551        }
552    }
553    super::precision::Component::new(nodes, shells, holes, source_order)
554}

556/// Partition by spatial connectivity before constructing any arrangement scratch.
557pub(crate) fn overlay<'a>(
558    a: impl IntoIterator<Item = &'a Polygon>,
559    b: impl IntoIterator<Item = &'a Polygon>,
560    intersection: bool,
561) -> super::precision::Geometry {
562    let polygons = a
563        .into_iter()
564        .map(|p| (p, 0))
565        .chain(b.into_iter().map(|p| (p, 1)))
566        .collect::<Vec<_>>();
567    if polygons.is_empty() {
568        return super::precision::Geometry::default();
569    }
570    let tree = index::build(
571        polygons
572            .iter()
573            .enumerate()
574            .filter_map(|(i, (p, _))| p.bounding_rect().map(|r| (i, r))),
575    );
576    let mut visited = vec![false; polygons.len()];
577    let mut output = Vec::new();
578    for seed in 0..polygons.len() {
579        crate::cancel::check();
580        if visited[seed] {
581            continue;
582        }
583        visited[seed] = true;
584        let mut queue = VecDeque::from([seed]);
585        let mut group = Vec::new();
586        while let Some(i) = queue.pop_front() {
587            group.push(polygons[i]);
588            if let Some(rect) = polygons[i].0.bounding_rect() {
589                for item in tree.locate_in_envelope_intersecting(&index::envelope(rect)) {
590                    if !visited[item.data] {
591                        visited[item.data] = true;
592                        queue.push_back(item.data);
593                    }
594                }
595            }
596        }
597        if !intersection || group.iter().any(|p| p.1 == 0) && group.iter().any(|p| p.1 == 1) {
598            output.push(component(&group, intersection));
599        }
600    }
601    super::precision::Geometry(output)
602}

604pub(crate) fn midpoint_inside(location: &Location, a: Coord, b: Coord) -> bool {
605    midpoint_position(location, a, b) == 1
606}
607pub(crate) fn midpoint_position(location: &Location, a: Coord, b: Coord) -> i8 {
608    if let Some(position) = certified_midpoint_position(location, a, b) {
609        return position;
610    }
611    let (ax, ay) = Exact::input(a).xy();
612    let (bx, by) = Exact::input(b).xy();
613    let two = Q::from_integer(2.into());
614    position_exact(
615        location,
616        &Exact::rational((ax + bx) / &two, (ay + by) / two),
617    )
618}

620fn certified_midpoint_position(location: &Location, a: Coord, b: Coord) -> Option<i8> {
621    use certified::Interval;
622    let a = certified::coordinates(&Exact::input(a));
623    let b = certified::coordinates(&Exact::input(b));
624    let p = [
625        a[0].add(b[0]).mul(Interval(0.5, 0.5)),
626        a[1].add(b[1]).mul(Interval(0.5, 0.5)),
627    ];
628    certified_position(location, p)
629}

631/// Ray parity of every point in a coordinate box, or `None` when the box may
632/// touch the boundary or straddle a decision.
633pub(super) fn certified_position(location: &Location, p: [certified::Interval; 2]) -> Option<i8> {
634    if p.iter().any(|i| !i.0.is_finite() || !i.1.is_finite()) {
635        return None;
636    }
637    let query = rstar::AABB::from_corners([p[0].0, p[1].0], [f64::MAX, p[1].1]);
638    let above = |y: f64| {
639        if y > p[1].1 {
640            Some(true)
641        } else if y <= p[1].0 {
642            Some(false)
643        } else {
644            None
645        }
646    };
647    let mut inside = false;
648    for item in location.tree.locate_in_envelope_intersecting(&query) {
649        crate::cancel::check();
650        let s = location.segments[item.data];
651        let bounds = s.bounding_rect();
652        let contact_possible = p[0].1 >= bounds.min().x
653            && p[0].0 <= bounds.max().x
654            && p[1].1 >= bounds.min().y
655            && p[1].0 <= bounds.max().y;
656        if s.start.y == s.end.y && !contact_possible {
657            continue;
658        }
659        let (Some(a_above), Some(b_above)) = (above(s.start.y), above(s.end.y)) else {
660            return None;
661        };
662        if a_above == b_above && !contact_possible {
663            continue;
664        }
665        let [ax, ay] = certified::coordinates(&Exact::input(s.start));
666        let [bx, by] = certified::coordinates(&Exact::input(s.end));
667        let side = bx
668            .sub(ax)
669            .mul(p[1].sub(ay))
670            .sub(by.sub(ay).mul(p[0].sub(ax)))
671            .sign();
672        let side = side?;
673        if a_above != b_above && side.is_gt() == b_above {
674            inside = !inside;
675        }
676    }
677    Some(if inside { 1 } else { -1 })
678}
