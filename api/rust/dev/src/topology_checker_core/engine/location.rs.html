1//! Reusable boundary and ray indexes over original polygon coordinates.
2use super::{
3    index::{self, Index},
4    proximity,
5};
6use geo::{
7    BoundingRect, Coord, Kernel, Line, Polygon, Rect, Winding,
8    kernels::{Orientation, RobustKernel},
9};
10
11#[derive(Debug)]
12pub(crate) struct Location {
13    pub segments: Vec<Line>,
14    pub tree: Index,
15    pub bounds: Rect,
16    pub anchors: Vec<(usize, usize)>,
17    pub interior_left: Vec<bool>,
18}
19impl Location {
20    pub fn new(polygon: &Polygon) -> Self {
21        let interior_left = std::iter::once(polygon.exterior())
22            .chain(polygon.interiors())
23            .enumerate()
24            .flat_map(|(i, r)| {
25                let left = (r.winding_order()
26                    == Some(geo::winding_order::WindingOrder::CounterClockwise))
27                    == (i == 0);
28                std::iter::repeat_n(left, r.0.len() - 1)
29            })
30            .collect();
31        let mut start = 0;
32        let anchors = std::iter::once(polygon.exterior())
33            .chain(polygon.interiors())
34            .map(|ring| {
35                let end = start + ring.0.len() - 1;
36                let range = (start, end);
37                start = end;
38                range
39            })
40            .collect();
41        let segments = std::iter::once(polygon.exterior())
42            .chain(polygon.interiors())
43            .flat_map(|r| r.lines())
44            .collect::<Vec<_>>();
45        let tree = index::build(
46            segments
47                .iter()
48                .enumerate()
49                .map(|(i, s)| (i, s.bounding_rect())),
50        );
51        Self {
52            segments,
53            tree,
54            bounds: polygon.bounding_rect().expect("clean polygon"),
55            anchors,
56            interior_left,
57        }
58    }
59    /// -1 outside, 0 boundary, 1 interior. Ray decisions use robust orientations.
60    pub fn position(&self, p: Coord) -> i8 {
61        if p.x < self.bounds.min().x
62            || p.x > self.bounds.max().x
63            || p.y < self.bounds.min().y
64            || p.y > self.bounds.max().y
65        {
66            return -1;
67        }
68        let query = index::envelope(Rect::new(
69            p,
70            Coord {
71                x: self.bounds.max().x,
72                y: p.y,
73            },
74        ));
75        let mut inside = false;
76        for item in self.tree.locate_in_envelope_intersecting(&query) {
77            let s = self.segments[item.data];
78            let side = RobustKernel::orient2d(s.start, s.end, p);
79            if side == Orientation::Collinear
80                && p.x >= s.start.x.min(s.end.x)
81                && p.x <= s.start.x.max(s.end.x)
82                && p.y >= s.start.y.min(s.end.y)
83                && p.y <= s.start.y.max(s.end.y)
84            {
85                return 0;
86            }
87            if (s.start.y > p.y) != (s.end.y > p.y)
88                && (side == Orientation::CounterClockwise) == (s.end.y > s.start.y)
89            {
90                inside = !inside;
91            }
92        }
93        if inside { 1 } else { -1 }
94    }
95    /// Segment `i` directed with the interior on its left, and the far end of
96    /// the segment that follows it in that direction.
97    pub fn edge(&self, i: usize) -> (Line, Coord) {
98        let (start, end) = self.anchors[self.anchors.partition_point(|&(s, _)| s <= i) - 1];
99        let s = self.segments[i];
100        if self.interior_left[i] {
101            (
102                s,
103                self.segments[if i + 1 == end { start } else { i + 1 }].end,
104            )
105        } else {
106            (
107                Line::new(s.end, s.start),
108                self.segments[if i == start { end - 1 } else { i - 1 }].start,
109            )
110        }
111    }
112    pub fn near(&self, p: Coord, tolerance: f64) -> bool {
113        self.tree
114            .locate_in_envelope_intersecting(&index::at(p, tolerance))
115            .any(|item| {
116                let s = self.segments[item.data];
117                if tolerance == 0. {
118                    RobustKernel::orient2d(s.start, s.end, p) == Orientation::Collinear
119                        && s.bounding_rect().contains_coord(p)
120                } else {
121                    proximity::near_segment(p, s, tolerance)
122                }
123            })
124    }
125}
126trait ContainsCoordinate {
127    fn contains_coord(&self, p: Coord) -> bool;
128}
129impl ContainsCoordinate for Rect {
130    fn contains_coord(&self, p: Coord) -> bool {
131        p.x >= self.min().x && p.x <= self.max().x && p.y >= self.min().y && p.y <= self.max().y
132    }
133}
134/// The location of each polygon part, built once. Each part builds an index of its
135/// own, so the build checks for cancellation before each.
136pub(crate) fn locations(parts: &crate::model::Parts) -> &[Location] {
137    parts.locations.get_or_init(|| {
138        parts
139            .polygons
140            .iter()
141            .map(|p| {
142                crate::cancel::checkpoint("locations");
143                Location::new(&p.geometry)
144            })
145            .collect()
146    })
