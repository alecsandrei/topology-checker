1//! Shared intervals are evidence from original segments, independent of clustering.
2use super::{cluster::distance, line_graph::projection};
3use geo::{Coord, Line, LineIntersection, line_intersection::line_intersection};
4
5pub(crate) fn point(line: Line, t: f64) -> Coord {
6    if t == 0. {
7        line.start
8    } else if t == 1. {
9        line.end
10    } else {
11        Coord {
12            x: line.start.x + t * (line.end.x - line.start.x),
13            y: line.start.y + t * (line.end.y - line.start.y),
14        }
15    }
16}
17/// The common longitudinal span must stay within tolerance at both ends.
18/// Linear separation then stays within tolerance everywhere on that span.
19/// Its length must exceed the transverse drift, excluding junction connectors.
20pub(crate) fn interval(a: Line, b: Line, tolerance: f64) -> Option<[[f64; 2]; 2]> {
21    evidence(a, b, tolerance).map(|e| e.spans)
22}
23
24pub(crate) struct Evidence {
25    pub spans: [[f64; 2]; 2],
26    pub support: [[f64; 2]; 2],
27    pub exact: bool,
28}

30pub(crate) fn evidence(a: Line, b: Line, tolerance: f64) -> Option<Evidence> {
31    let intersection = line_intersection(a, b);
32    if tolerance == 0. {
33        return match intersection {
34            Some(LineIntersection::Collinear { intersection })
35                if intersection.start != intersection.end =>
36            {
37                let span = |s| {
38                    let x = projection(s, intersection.start).0;
39                    let y = projection(s, intersection.end).0;
40                    [x.min(y), x.max(y)]
41                };
42                let spans = [span(a), span(b)];
43                Some(Evidence {
44                    spans,
45                    support: spans,
46                    exact: true,
47                })
48            }
49            _ => None,
50        };
51    }
52    let endpoint_contact = matches!(
53        intersection,
54        Some(LineIntersection::SinglePoint {
55            is_proper: false,
56            ..
57        })
58    );
59    let vector = |s: Line| (s.end.x - s.start.x, s.end.y - s.start.y);
60    let (ax, ay) = vector(a);
61    let (bx, by) = vector(b);
62    let al = ax.hypot(ay);
63    let bl = bx.hypot(by);
64    if al == 0. || bl == 0. {
65        return None;
66    }
67    // A canonical longitudinal direction makes operand and path reversal invariant.
68    let choose_a = al > bl || al == bl && (ax.abs(), ay.abs()) >= (bx.abs(), by.abs());
69    let (mut ux, mut uy) = if choose_a {
70        (ax / al, ay / al)
71    } else {
72        (bx / bl, by / bl)
73    };
74    if ux < 0. || ux == 0. && uy < 0. {
75        ux = -ux;
76        uy = -uy;
77    }
78    let origin = if (a.start.x, a.start.y) < (b.start.x, b.start.y) {
79        a.start
80    } else {
81        b.start
82    };
83    let along = |p: Coord| (p.x - origin.x) * ux + (p.y - origin.y) * uy;
84    let av = [along(a.start), along(a.end)];
85    let bv = [along(b.start), along(b.end)];
86    let lo = av[0].min(av[1]).max(bv[0].min(bv[1]));
87    let hi = av[0].max(av[1]).min(bv[0].max(bv[1]));
88    if hi <= lo || av[0] == av[1] || bv[0] == bv[1] {
89        return None;
90    }
91    let span = |v: [f64; 2]| {
92        let x = ((lo - v[0]) / (v[1] - v[0])).clamp(0., 1.);
93        let y = ((hi - v[0]) / (v[1] - v[0])).clamp(0., 1.);
94        [x.min(y), x.max(y)]
95    };
96    let (mut sa, mut sb) = (span(av), span(bv));
97    let support = [sa, sb];
98    let at = |s: Line, v: [f64; 2], t: f64| point(s, ((t - v[0]) / (v[1] - v[0])).clamp(0., 1.));
99    let (a0, a1, b0, b1) = (at(a, av, lo), at(a, av, hi), at(b, bv, lo), at(b, bv, hi));
100    if distance(a0, b0) > tolerance || distance(a1, b1) > tolerance {
101        return None;
102    }
103    // Compute change from original vectors. Rounded projected coordinates can
104    // coincide at a short junction span and falsely erase transverse drift.
105    let drift = (hi - lo)
106        * (ax / (av[1] - av[0]) - bx / (bv[1] - bv[0]))
107            .hypot(ay / (av[1] - av[0]) - by / (bv[1] - bv[0]));
108    if drift >= hi - lo {
109        return None;
110    }
111    // A junction needs sustained support beyond its tolerance neighbourhood.
112    // Subtract transverse drift before testing the common longitudinal extent,
113    // and do so before extending endpoint caps. Long near-duplicate passages
114    // sharing source vertices still qualify, including unequal segmentation.
115    if endpoint_contact && hi - lo <= tolerance + drift {
116        return None;
117    }
118    // Short endpoint caps are covered only by an actual nearby target segment.
119    // This also retains original vertices instead of replacing them by projections.
120    for (s, other, range) in [(a, b, &mut sa), (b, a, &mut sb)] {
121        if distance(s.start, projection(other, s.start).1) <= tolerance {
122            range[0] = 0.;
123        }
124        if distance(s.end, projection(other, s.end).1) <= tolerance {
125            range[1] = 1.;
126        }
127    }
128    Some(Evidence {
129        spans: [sa, sb],
130        support,
131        exact: matches!(intersection, Some(LineIntersection::Collinear { intersection }) if intersection.start != intersection.end),
132    })
133}

135pub(crate) fn union(spans: Vec<[f64; 2]>) -> Vec<[f64; 2]> {
136    union_within(spans, 0.)
137}
138/// The union of spans, joining those at most `gap` apart.
139pub(crate) fn union_within(mut spans: Vec<[f64; 2]>, gap: f64) -> Vec<[f64; 2]> {
140    spans.sort_by(|a, b| a[0].total_cmp(&b[0]).then(a[1].total_cmp(&b[1])));
141    let mut result: Vec<[f64; 2]> = Vec::new();
142    for span in spans {
143        if let Some(last) = result.last_mut()
144            && span[0] <= last[1] + gap
145        {
146            last[1] = last[1].max(span[1]);
147        } else {
148            result.push(span);
149        }
150    }
151    result
152}
