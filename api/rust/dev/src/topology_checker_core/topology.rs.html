16/// One rule's violations, with deterministic counts and feature provenance.
17#[derive(Debug, Clone, Serialize)]
18#[non_exhaustive]
19pub struct RuleReport {
20    /// Registered rule with its topology-scoped rule and layer IDs.
21    pub spec: RuleSpec,
22    /// Primary input layer name.
23    pub layer: String,
24    /// Secondary input layer name for a binary rule.
25    pub other: Option<String>,
26    /// Number of aggregated violations per kind, in kind order.
27    pub counts: BTreeMap<ViolationKind, usize>,
28    /// Deterministically ordered violations, serialized as `errors`.
29    /// Several parts may contribute to one violation.
30    #[serde(rename = "errors")]
31    pub violations: Vec<Violation>,
32}
33impl RuleReport {
34    /// Return the name of one of this rule's input layers, as referenced by its violations.
35    #[must_use]
36    pub fn layer_name(&self, id: LayerId) -> Option<&str> {
37        if id == self.spec.layer {
38            Some(&self.layer)
39        } else if Some(id) == self.spec.other {
40            self.other.as_deref()
41        } else {
42            None
43        }
44    }
45}
46/// A non-fatal input condition found while preparing a registered layer.
47#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
48#[serde(tag = "kind", rename_all = "kebab-case")]
49#[non_exhaustive]
50pub enum Warning {
51    /// A cleaned line part collapsed under the configured tolerance.
52    CollapsedLine {
53        /// Topology-scoped layer ID.
54        layer: LayerId,
55        /// Caller-supplied layer name.
56        name: String,
57        /// Original feature ID.
58        fid: i64,
59        /// Original part index.
60        part: u32,
61    },
62    /// Degenerate parts or rings were skipped while cleaning a layer.
63    DegenerateParts {
64        /// Topology-scoped layer ID.
65        layer: LayerId,
66        /// Caller-supplied layer name, which need not be unique.
67        name: String,
68        /// Number of skipped parts and rings.
69        skipped: usize,
70    },
71}
72impl fmt::Display for Warning {
73    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
74        match self {
75            Self::CollapsedLine {
76                name, fid, part, ..
77            } => write!(
78                f,
79                "layer {name}, FID {fid}, part {part}: line collapsed under tolerance"
80            ),
81            Self::DegenerateParts { name, skipped, .. } => {
82                write!(
83                    f,
84                    "layer {name}: skipped {skipped} degenerate parts or rings"
85                )
86            }
87        }
88    }
89}
90/// A successful validation result, which may contain topology violations.
91#[derive(Debug, Clone, Serialize)]
92#[non_exhaustive]
93pub struct Report {
94    /// Planar tolerance in input coordinate units.
95    pub tolerance: f64,
96    /// Degenerate-part warnings in layer registration order.
97    pub warnings: Vec<Warning>,
98    /// Rule results in rule registration order, independent of parallel scheduling.
99    /// The result for `RuleId` `i` is at index `i`.
100    pub rules: Vec<RuleReport>,
101}
102impl Report {
103    /// Return whether every registered rule has no violations.
104    #[must_use]
105    pub fn is_valid(&self) -> bool {
106        self.rules.iter().all(|r| r.violations.is_empty())
107    }
108    /// Count aggregated violations across all registered rules.
109    #[must_use]
110    pub fn violation_count(&self) -> usize {
111        self.rules.iter().map(|r| r.violations.len()).sum()
112    }
113    /// Iterate violations in rule order and then each rule's deterministic order.
114    pub fn violations(&self) -> impl Iterator<Item = &Violation> {
115        self.rules.iter().flat_map(|r| &r.violations)
116    }
117    /// Return one rule's result. IDs must originate from the topology that
118    /// produced this report.
119    #[must_use]
120    pub fn rule(&self, id: RuleId) -> Option<&RuleReport> {
121        self.rules.get(id.0)
122    }
123}

133/// Shared planar layers and ordered rules with reusable, thread-safe geometry caches.
134#[derive(Debug)]
135pub struct Topology {
136    tolerance: f64,
137    cancellation: Option<crate::CancellationToken>,
138    selection: Option<(LayerId, i64, Arc<Parts>)>,
139    layers: Vec<Arc<Layer>>,
140    specs: Vec<RuleSpec>,
141    graphs: GraphCache,
142}
143impl Topology {
144    /// Create an empty topology with tolerance in input coordinate units.
145    /// Zero requests exact comparisons; no CRS transformation is performed.
146    ///
147    /// # Errors
148    /// Returns [`Error::Tolerance`] for a negative or non-finite tolerance.
149    pub fn new(tolerance: f64) -> Result<Self, Error> {
150        if !tolerance.is_finite() || tolerance < 0. {
151            return Err(Error::Tolerance(tolerance));
152        }
153        Ok(Self {
154            tolerance,
155            cancellation: None,
156            selection: None,
157            layers: Vec::new(),
158            specs: Vec::new(),
159            graphs: Arc::new(Mutex::new(BTreeMap::new())),
160        })
161    }
162    /// Return the configured planar tolerance.
163    #[must_use]
164    pub fn tolerance(&self) -> f64 {
165        self.tolerance
166    }
167    /// Register a layer without validating it, returning a topology-scoped ID.
168    /// Pass an `Arc<Layer>` to share one layer, and its cached views, between topologies.
169    pub fn add_layer(&mut self, layer: impl Into<Arc<Layer>>) -> LayerId {
170        let id = LayerId(self.layers.len());
171        self.layers.push(layer.into());
172        id
173    }
174    /// Borrow a registered layer. IDs must originate from this topology.
175    ///
176    /// # Errors
177    /// Returns [`Error::UnknownLayer`] when the index is not registered.
178    pub fn layer(&self, id: LayerId) -> Result<&Layer, Error> {
179        self.layers
180            .get(id.0)
181            .map(AsRef::as_ref)
182            .ok_or(Error::UnknownLayer(id))
183    }
184    /// Share a registered immutable layer and its caches.
185    ///
186    /// # Errors
187    /// Returns an unknown-layer error for an unregistered ID.
188    pub fn shared_layer(&self, id: LayerId) -> Result<Arc<Layer>, Error> {
189        self.layers
190            .get(id.0)
191            .map(Arc::clone)
192            .ok_or(Error::UnknownLayer(id))
193    }
194    /// Iterate layers and their IDs in registration order.
195    pub fn layers(&self) -> impl Iterator<Item = (LayerId, &Layer)> {
196        self.layers
197            .iter()
198            .enumerate()
199            .map(|(i, l)| (LayerId(i), l.as_ref()))
200    }
201    /// Iterate registered rules in registration order.
202    pub fn rules(&self) -> impl ExactSizeIterator<Item = RuleSpec> {
203        self.specs.iter().copied()
204    }
205    /// Remove every registered rule, keeping layers and their cached line graphs, so a
206    /// different rule set can be validated without rebuilding them. Rule IDs restart at
207    /// zero; IDs from earlier registrations must not be reused.
208    pub fn clear_rules(&mut self) {
209        self.specs.clear();
210    }
211    pub(crate) fn parts(&self, id: LayerId) -> Result<&Parts, Error> {
212        if let Some((layer, _, parts)) = &self.selection
213            && *layer == id
214        {
215            return Ok(parts);
216        }
217        self.full_parts(id)
218    }
219    pub(crate) fn select_feature(&mut self, layer: LayerId, fid: i64) -> Result<(), Error> {
220        let parts = Arc::new(self.full_parts(layer)?.feature(fid));
221        self.selection = Some((layer, fid, parts));
222        Ok(())
223    }
224    pub(crate) fn selected_feature(&self) -> Option<(LayerId, i64)> {
225        self.selection
226            .as_ref()
227            .map(|(layer, fid, _)| (*layer, *fid))
228    }
229    pub(crate) fn feature_matches(&self, layer: LayerId, fid: i64) -> bool {
230        self.selected_feature()
231            .is_none_or(|selected| selected == (layer, fid))
232    }
233    pub(crate) fn full_parts(&self, id: LayerId) -> Result<&Parts, Error> {
234        let layer = self.layer(id)?;
235        layer.parts().map_err(|e| e.at(id, layer.name()))
236    }
237    pub(crate) fn graph(&self, layers: Vec<LayerId>) -> Result<Arc<LineGraph>, Error> {
238        self.graph_inputs(
239            layers
240                .into_iter()
241                .map(|id| (id, InputMode::Linework))
242                .collect(),
243        )
244    }
245    pub(crate) fn graph_inputs(
246        &self,
247        mut layers: Vec<(LayerId, InputMode)>,
248    ) -> Result<Arc<LineGraph>, Error> {
249        layers.sort_unstable();
250        layers.dedup();
251        // A poisoned map still contains valid Arc/Cache entries: initialization
252        // happens outside this lock, and no partially built graph is stored in it.
253        let slot = Arc::clone(
254            self.graphs
255                .lock()
256                .unwrap_or_else(std::sync::PoisonError::into_inner)
257                .entry(layers.clone())
258                .or_default(),
259        );
260        slot.get_or_init(|| {
261            let mut full = self.snapshot();
262            full.selection = None;
263            LineGraph::build(&full, &layers).map(Arc::new)
264        })
265        .as_ref()
266        .map(Arc::clone)
267        .map_err(Clone::clone)
268    }
269    /// Register a rule after validating input families, finite coordinates and FIDs.
270    /// Unary rules require no secondary layer; binary rules require a distinct one.
271    /// Rules run in parallel, but reports retain registration order.
272    /// The returned spec's `id` locates this rule's result with [`Report::rule`].
273    ///
274    /// # Errors
275    /// Returns input-count, same-layer, unknown-layer, geometry-kind, duplicate-FID
276    /// or non-finite-coordinate errors. Failed registration does not add a rule.
277    pub fn add_rule(
278        &mut self,
279        rule: Rule,
280        layer: LayerId,
281        other: Option<LayerId>,
282    ) -> Result<RuleSpec, Error> {
283        self.register_rule(
284            rule,
285            LayerInput::Existing(layer),
286            other.map(LayerInput::Existing),
287        )
288    }
289    /// Validate and commit owned inputs and a rule as one transaction.
290    /// Layer caches do not depend on registration IDs, so validation work done
291    /// here is reused after commit.
292    ///
293    /// # Errors
294    /// Returns arity, same-layer, unknown-layer, geometry-kind, duplicate-FID or
295    /// coordinate errors before committing any new layer or rule.
296    pub fn register_rule(
297        &mut self,
298        rule: Rule,
299        layer: LayerInput,
300        other: Option<LayerInput>,
301    ) -> Result<RuleSpec, Error> {
302        let (primary_kind, other_kind) = rule.inputs();
303        rule.check_input_count(other.is_some())?;
304        if let (LayerInput::Existing(a), Some(LayerInput::Existing(b))) = (&layer, &other)
305            && a == b
306        {
307            return Err(Error::SameLayer { rule, layer: *a });
308        }
309        let mut next = self.layers.len();
310        let mut check = |input: &LayerInput, kind| -> Result<LayerId, Error> {
311            let (id, value) = match input {
312                LayerInput::Existing(id) => (*id, self.layer(*id)?),
313                LayerInput::New(value) => {
314                    let id = LayerId(next);
315                    next += 1;
316                    (id, value.as_ref())
317                }
318            };
319            value.check_kind(id, kind)?;
320            value.parts().map_err(|e| e.at(id, value.name()))?;
321            Ok(id)
322        };
323        let primary_id = check(&layer, primary_kind)?;
324        let other_id = other
325            .as_ref()
326            .zip(other_kind)
327            .map(|(input, kind)| check(input, kind))
328            .transpose()?;
329        for input in [Some(layer), other].into_iter().flatten() {
330            if let LayerInput::New(value) = input {
331                self.layers.push(Arc::from(value));
332            }
333        }
334        let spec = RuleSpec {
335            id: RuleId(self.specs.len()),
336            rule,
337            layer: primary_id,
338            other: other_id,
339        };
340        self.specs.push(spec);
341        Ok(spec)
342    }
343    /// Validate cached part views and report skipped degenerate parts without running rules.
344    ///
345    /// # Errors
346    /// Returns duplicate-FID or non-finite-coordinate errors from any registered layer,
347    /// including layers that are not used by a rule.
348    pub fn warnings(&self) -> Result<Vec<Warning>, Error> {
349        let mut warnings = Vec::new();
350        for (id, layer) in self.layers() {
351            let parts = self.parts(id)?;
352            for line in &parts.lines {
353                if crate::engine::line_graph::collapsed(&line.geometry, self.tolerance) {
354                    warnings.push(Warning::CollapsedLine {
355                        layer: id,
356                        name: layer.name().into(),
357                        fid: line.reference.fid,
358                        part: line.reference.part.unwrap_or(0),
359                    });
360                }
361            }
362            let skipped = parts.skipped;
363            if skipped > 0 {
364                warnings.push(Warning::DegenerateParts {
365                    layer: id,
366                    name: layer.name().into(),
367                    skipped,
368                });
369            }
370        }
371        Ok(warnings)
372    }
373    /// Capture immutable layers, rules and shared caches for independent validation.

376        Self {
377            tolerance: self.tolerance,
378            cancellation: self.cancellation.clone(),
379            selection: self.selection.clone(),
380            layers: self.layers.clone(),
381            specs: self.specs.clone(),
382            graphs: Arc::clone(&self.graphs),

385    /// Send aggregated violations to a consumer in rule, kind and feature order.
386    /// The consumer receives the original registered rule specification. Returning
387    /// an error stops output immediately; successful exhaustion returns metadata.
388    /// Duplicate point stacks require memory proportional to inputs, not pair count.
389    ///
390    /// # Errors
391    /// Returns deferred input failures or the consumer's original error.
392    pub fn validate_into<E>(
393        &self,
394        mut sink: impl FnMut(RuleSpec, Violation) -> Result<(), E>,
395    ) -> Result<crate::ReportSummary, crate::SinkError<E>> {
396        let warnings = self.warnings()?;
397        let mut summaries = Vec::new();
398        for &spec in &self.specs {
399            let mut counts = BTreeMap::new();
400            let mut total = 0;
401            crate::cancel::check();
402            rules::run_into(self, spec, |v| {
403                crate::cancel::check();
404                let kind = v.kind;
405                sink(spec, v)?;
406                *counts.entry(kind).or_default() += 1;
407                total += 1;
408                Ok(())
409            })?;
410            crate::cancel::check();
411            summaries.push(crate::RuleSummary {
412                spec,
413                layer: self.layer(spec.layer)?.name().into(),
414                other: spec
415                    .other
416                    .map(|id| self.layer(id).map(|l| l.name().into()))
417                    .transpose()?,
418                counts,
419                total,
420            });
421        }
422        Ok(crate::ReportSummary {
423            tolerance: self.tolerance,
424            warnings,
425            rules: summaries,
426        })
427    }
428    /// Validate with cooperative cancellation, retaining the original sink error.
429    ///
430    /// # Errors
431    /// Returns input failures, cancellation, or the consumer's original error.
432    pub fn validate_into_cancellable<E>(
433        &self,
434        token: crate::CancellationToken,
435        sink: impl FnMut(RuleSpec, Violation) -> Result<(), E>,
436    ) -> Result<crate::ReportSummary, crate::SinkError<E>> {
437        let mut snapshot = self.snapshot();
438        snapshot.cancellation = Some(token.clone());
439        match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
440            crate::cancel::scope(Some(token), || snapshot.validate_into(sink))
441        })) {
442            Ok(result) => result,
443            Err(error) if error.is::<crate::cancel::Cancelled>() => {
444                Err(crate::SinkError::Core(Error::Cancelled))
445            }
446            Err(error) => std::panic::resume_unwind(error),
447        }
448    }
449    pub(crate) fn cancellable<T>(&self, run: impl FnOnce() -> T) -> T {
450        crate::cancel::scope(self.cancellation.clone(), run)
451    }
452    /// Validate all layers and execute registered rules, reusing cached views.
453    /// Topology violations are a successful report, not operational errors.
454    ///
455    /// # Errors
456    /// Returns invalid-layer errors from deferred validation or operational errors
457    /// encountered while building rule inputs. No partial report is returned.
458    pub fn validate(&self) -> Result<Report, Error> {
459        let warnings = self.warnings()?;
460        let rules = self
461            .specs
462            .par_iter()
463            .map(|&spec| {
464                let mut violations = rules::run(self, spec)?;
465                violations.sort_by_key(|v| {
466                    (
467                        v.kind,
468                        v.origin.map(crate::FeatureRef::feature),
469                        v.destination.map(crate::FeatureRef::feature),
470                    )
471                });
472                let mut counts = BTreeMap::new();
473                for v in &violations {
474                    *counts.entry(v.kind).or_default() += 1;
475                }
476                Ok(RuleReport {
477                    spec,
478                    layer: self.layer(spec.layer)?.name().into(),
479                    other: spec
480                        .other
481                        .map(|id| self.layer(id).map(|l| l.name().to_string()))
482                        .transpose()?,
483                    counts,
484                    violations,
485                })
486            })
487            .collect::<Result<Vec<_>, Error>>()?;
488        Ok(Report {
489            tolerance: self.tolerance,
490            warnings,
491            rules,
492        })
493    }

501    #[test]
502    fn transactions_reject_invalid_layers_without_consuming_ids_or_rules() {
503        for bad in [
504            Layer::from_features("duplicates", vec![(4, Point::new(0., 0.).into()); 2]),
505            Layer::from_geometries("nonfinite", [Point::new(f64::INFINITY, 0.).into()]),
506            Layer::from_geometries(
507                "wrong kind",
508                [LineString::from(vec![(0., 0.), (1., 0.)]).into()],
509            ),
510        ] {
511            let mut topology = Topology::new(0.).unwrap();
512            let original = topology.add_layer(Layer::from_geometries("original", []));
513            let new = LayerInput::New(Box::new(Layer::from_geometries(
514                "pending",
515                [Point::new(0., 0.).into()],
516            )));
517            assert!(
518                topology
519                    .register_rule(
520                        Rule::PointMustNotOverlapWith,
521                        new,
522                        Some(LayerInput::New(Box::new(bad))),
523                    )
524                    .is_err()
525            );
526            assert_eq!(topology.layers().count(), 1);
527            assert!(topology.validate().unwrap().rules.is_empty());
528            assert!(topology.warnings().unwrap().is_empty());
529            let spec = topology
530                .register_rule(
531                    Rule::PointMustNotOverlapWith,
532                    LayerInput::Existing(original),
533                    Some(LayerInput::New(Box::new(Layer::from_geometries(
534                        "next",
535                        [],
536                    )))),
537                )
538                .unwrap();
539            assert_eq!(spec.other.unwrap().index(), 1);
540        }
541    }

543    #[test]
544    fn transaction_caches_keep_final_ids_for_each_input_order() {
545        for reuse_primary in [false, true] {
546            let mut topology = Topology::new(0.).unwrap();
547            topology.add_layer(Layer::from_geometries("unused", []));
548            let existing = topology.add_layer(Layer::from_geometries(
549                "existing",
550                [Point::new(0., 0.).into()],
551            ));
552            let fresh = LayerInput::New(Box::new(Layer::from_geometries(
553                "fresh",
554                [Point::new(0., 0.).into()],
555            )));
556            let (primary, other) = if reuse_primary {
557                (LayerInput::Existing(existing), fresh)
558            } else {
559                (fresh, LayerInput::Existing(existing))
560            };
561            let spec = topology
562                .register_rule(Rule::PointMustNotOverlapWith, primary, Some(other))
563                .unwrap();
564            let report = topology.validate().unwrap();
565            let error = &report.rules[0].violations[0];
566            assert_eq!(error.origin.unwrap().layer, spec.layer);
567            assert_eq!(error.destination.unwrap().layer, spec.other.unwrap());
568            assert_eq!(topology.layers().count(), 3);
569        }
570    }
