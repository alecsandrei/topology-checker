3use geo::{
4    Geometry, GeometryCollection, LineString, MultiLineString, MultiPoint, MultiPolygon, Point,
5    Polygon,
6};
7use std::collections::BTreeMap;

9/// How the geometries of merged violations are combined.
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub(crate) enum Merge {
12    /// Keep only the highest dimension present, such as overlap areas over their edges.
13    /// An unsupported first geometry, such as an empty multi geometry, is kept when
14    /// nothing else contributes.
15    #[cfg(test)]
16    HighestDimension,
17    /// Keep every dimension, such as isolated contacts alongside line pieces, in a
18    /// collection when several are present.
19    AllDimensions,
20}
21impl Merge {
22    /// Combine geometries. Coincident points are deduplicated.
23    pub(crate) fn geometry(self, geometries: impl IntoIterator<Item = Geometry>) -> Geometry {
24        let mut collected = Collected::default();
25        match self {
26            #[cfg(test)]
27            Self::HighestDimension => {
28                let mut fallback = None;
29                for (i, g) in geometries.into_iter().enumerate() {
30                    if let Some(unsupported) = collected.simple(g)
31                        && i == 0
32                    {
33                        fallback = Some(unsupported);
34                    }
35                }
36                let [points, lines, polygons] = collected.dimensions();
37                polygons
38                    .or(lines)
39                    .or(points)
40                    .or(fallback)
41                    .unwrap_or_else(|| MultiPoint(Vec::new()).into())
42            }
43            Self::AllDimensions => {
44                for g in geometries {
45                    collected.all(g);
46                }
47                let mut present = collected
48                    .dimensions()
49                    .into_iter()
50                    .flatten()
51                    .collect::<Vec<_>>();
52                if present.len() == 1 {
53                    present.remove(0)
54                } else {
55                    Geometry::GeometryCollection(GeometryCollection(present))
56                }
57            }
58        }
59    }
60}

62/// Points, lines and polygons gathered from several geometries.
63#[derive(Default)]
64struct Collected {
65    points: Vec<Point>,
66    lines: Vec<LineString>,
67    polygons: Vec<Polygon>,
68}
69impl Collected {
70    /// Collect a nonempty simple or multi geometry, returning any other geometry unchanged.
71    fn simple(&mut self, g: Geometry) -> Option<Geometry> {
72        match g {
73            Geometry::Point(g) => self.points.push(g),
74            Geometry::MultiPoint(g) if !g.0.is_empty() => self.points.extend(g.0),
75            Geometry::Line(g) => self.lines.push(g.into()),
76            Geometry::LineString(g) => self.lines.push(g),
77            Geometry::MultiLineString(g) if !g.0.is_empty() => self.lines.extend(g.0),
78            Geometry::Polygon(g) => self.polygons.push(g),
79            Geometry::MultiPolygon(g) if !g.0.is_empty() => self.polygons.extend(g.0),
80            other => return Some(other),
81        }
82        None
83    }
84    /// Collect every part, converting rectangles and triangles and flattening collections.
85    fn all(&mut self, g: Geometry) {
86        match g {
87            Geometry::Rect(g) => self.polygons.push(g.to_polygon()),
88            Geometry::Triangle(g) => self.polygons.push(g.to_polygon()),
89            Geometry::GeometryCollection(g) => {
90                for child in g {
91                    self.all(child);
92                }
93            }
94            // Empty multi geometries contribute nothing.
95            g => {
96                let _ = self.simple(g);
97            }
98        }
99    }
100    /// Return deduplicated points, then lines, then polygons, each when present.
101    fn dimensions(mut self) -> [Option<Geometry>; 3] {
102        fn single_or_multi<T: Into<Geometry>, M: Into<Geometry>>(
103            mut parts: Vec<T>,
104            multi: fn(Vec<T>) -> M,
105        ) -> Option<Geometry> {
106            match parts.len() {
107                0 => None,
108                1 => Some(parts.remove(0).into()),
109                _ => Some(multi(parts).into()),
110            }
111        }
112        // Polygon rule aggregation occurs on original operands before emission.
113        // Never feed emitted intersection coordinates into another overlay.
114        // Parallel candidate discovery must not change multipart ordering.
115        self.lines.sort_by(|a, b| {
116            a.0.iter()
117                .zip(&b.0)
118                .map(|(a, b)| a.x.total_cmp(&b.x).then(a.y.total_cmp(&b.y)))
119                .find(|order| !order.is_eq())
120                .unwrap_or_else(|| a.0.len().cmp(&b.0.len()))
121        });
122        self.points
123            .sort_by(|a, b| a.x().total_cmp(&b.x()).then(a.y().total_cmp(&b.y())));
124        self.points.dedup();
125        [
126            single_or_multi(self.points, MultiPoint),
127            single_or_multi(self.lines, MultiLineString),
128            single_or_multi(self.polygons, MultiPolygon),
129        ]
130    }
131}

133/// Merge violations sharing a rule, kind and origin and destination features.
134/// Part indices are cleared when several parts contributed; groups follow key order.
135pub(crate) fn aggregate(violations: Vec<Violation>, merge: Merge) -> Vec<Violation> {
136    type Key = (
137        Rule,
138        ViolationKind,
139        Option<(LayerId, i64)>,
140        Option<(LayerId, i64)>,
141    );
142    let mut groups: BTreeMap<Key, Vec<Violation>> = BTreeMap::new();
143    for v in violations {
144        groups
145            .entry((
146                v.rule,
147                v.kind,
148                v.origin.map(FeatureRef::feature),
149                v.destination.map(FeatureRef::feature),
150            ))
151            .or_default()
152            .push(v);
153    }
154    groups
155        .into_values()
156        .map(|group| {
157            let mut group = group.into_iter();
158            let mut first = group.next().expect("aggregation groups are nonempty");
159            let mut geometries = vec![std::mem::replace(
160                &mut first.geometry,
161                Geometry::GeometryCollection(GeometryCollection(Vec::new())),
162            )];
163            for v in group {
164                if let (Some(a), Some(b)) = (&mut first.origin, v.origin) {
165                    a.aggregate(b);
166                }
167                if let (Some(a), Some(b)) = (&mut first.destination, v.destination) {
168                    a.aggregate(b);
169                }
170                geometries.push(v.geometry);
171            }
172            first.geometry = merge.geometry(geometries);
173            first
174        })
175        .collect()
176}

178#[cfg(test)]
179mod tests {
180    use super::*;
181    use geo::{Rect, coord};
182
183    fn violation(geometry: Geometry, part: u32) -> Violation {
184        Violation {
185            rule: Rule::LineMustNotIntersect,
186            kind: ViolationKind::Crossing,
187            geometry,
188            origin: Some(FeatureRef {
189                layer: LayerId(0),
190                fid: -4,
191                part: Some(part),
192            }),
193            destination: Some(FeatureRef {
194                layer: LayerId(1),
195                fid: 8,
196                part: Some(2),
197            }),
198        }
199    }
200
201    #[test]
202    fn aggregation_retains_provenance_deduplication_and_geometry_priority() {
203        let point = Point::new(1., 0.);
204        let points = aggregate(
205            vec![violation(point.into(), 0), violation(point.into(), 1)],
206            Merge::HighestDimension,
207        );
208        assert_eq!(points.len(), 1);
209        assert_eq!(points[0].geometry, Geometry::Point(point));
210        assert_eq!(points[0].origin.unwrap().part, None);
211        assert_eq!(points[0].destination.unwrap().part, Some(2));
212        let line = LineString::from(vec![(0., 0.), (1., 0.)]);
213        let polygon = Rect::new(coord! {x:0.,y:0.}, coord! {x:1.,y:1.}).to_polygon();
214        let mixed = aggregate(
215            vec![
216                violation(point.into(), 0),
217                violation(line.into(), 0),
218                violation(polygon.clone().into(), 1),
219            ],
220            Merge::HighestDimension,
221        );
222        assert_eq!(mixed[0].geometry, Geometry::Polygon(polygon));
223        let empty = Geometry::MultiLineString(MultiLineString(Vec::new()));
224        assert_eq!(
225            aggregate(vec![violation(empty.clone(), 0)], Merge::HighestDimension)[0].geometry,
226            empty
227        );
228    }

230    #[test]
231    fn all_dimensions_keep_contacts_beside_pieces_and_flatten_collections() {
232        let point = Point::new(1., 0.);
233        let line = LineString::from(vec![(0., 0.), (1., 0.)]);
234        let rect = Rect::new(coord! {x:0.,y:0.}, coord! {x:1.,y:1.});
235        let merged = aggregate(
236            vec![
237                violation(point.into(), 0),
238                violation(
239                    Geometry::GeometryCollection(GeometryCollection(vec![
240                        point.into(),
241                        line.clone().into(),
242                    ])),
243                    1,
244                ),
245                violation(rect.into(), 1),
246            ],
247            Merge::AllDimensions,
248        );
249        assert_eq!(merged.len(), 1);
250        assert_eq!(
251            merged[0].geometry,
252            Geometry::GeometryCollection(GeometryCollection(vec![
253                point.into(),
254                line.clone().into(),
255                rect.to_polygon().into()
256            ]))
257        );
258        assert_eq!(
259            Merge::AllDimensions.geometry([line.clone().into()]),
260            Geometry::LineString(line)
261        );
262        assert_eq!(
263            Merge::AllDimensions.geometry([MultiPoint(Vec::new()).into()]),
264            Geometry::GeometryCollection(GeometryCollection(Vec::new()))
265        );
266    }

268    #[test]
269    fn convenience_geometries_collect_as_lines_and_polygons() {
270        let line = geo::Line::new(coord! {x: 0., y: 0.}, coord! {x: 1., y: 0.});
271        let triangle = geo::Triangle::new(
272            coord! {x: 0., y: 0.},
273            coord! {x: 1., y: 0.},
274            coord! {x: 0., y: 1.},
275        );
276        let merged = Merge::AllDimensions.geometry([line.into(), triangle.into()]);
277        assert_eq!(
278            merged,
279            Geometry::GeometryCollection(GeometryCollection(vec![
280                Geometry::LineString(line.into()),
281                Geometry::Polygon(triangle.to_polygon()),
282            ]))
283        );
284        let single = Merge::HighestDimension.geometry([line.into()]);
285        assert_eq!(single, Geometry::LineString(line.into()));
