13/// Which noded linework must, or must not, coincide with the other layer's.
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub(super) enum Boundary {
16    /// Line pieces must lie on polygon boundaries.
17    LinesCovered,
18    /// Polygon boundary pieces must lie on lines.
19    BoundariesCovered,
20    /// Line pieces must not lie on polygon boundaries.
21    NotShared,
22}
23
24/// Report primary pieces longer than tolerance that violate the coverage requirement.
25pub(super) fn linework(
26    t: &Topology,
27    spec: RuleSpec,
28    other: LayerId,
29    mode: Boundary,
30) -> Result<Vec<Violation>, Error> {
31    if mode != Boundary::NotShared {
32        return uncovered(t, spec, other, mode);
33    }
34    let (primary, secondary) = if mode == Boundary::BoundariesCovered {
35        (InputMode::PolygonBoundary, InputMode::Linework)
36    } else {
37        (InputMode::Linework, InputMode::PolygonBoundary)
38    };
39    let graph = t.graph_inputs(vec![(spec.layer, primary), (other, secondary)])?;
40    let overlap = mode == Boundary::NotShared;
41    let mut coverage = vec![Vec::new(); graph.parts.len()];
42    for m in graph.matches() {
43        let [a, b] = m.sources.map(|s| graph.segment(s));
44        let (ar, br) = (graph.parts[a.part].reference, graph.parts[b.part].reference);
45        if ar.layer == br.layer {
46            continue;
47        }
48        for (s, span, target) in [(a, m.spans[0], br), (b, m.spans[1], ar)] {
49            coverage[s.part].push((s.start as f64 + span[0], s.start as f64 + span[1], target));
50        }
51    }
52    let mut violations = Vec::new();
53    for (part_id, part) in graph.parts.iter().enumerate() {
54        crate::cancel::check();
55        if part.reference.layer != spec.layer
56            || t.selected_feature()
57                .is_some_and(|key| key != part.reference.feature())
58        {
59            continue;
60        }
61        let targets = &coverage[part_id];
62        let destinations = if overlap {
63            targets
64                .iter()
65                .map(|x| x.2)
66                .collect::<BTreeSet<_>>()
67                .into_iter()
68                .map(Some)
69                .collect::<Vec<_>>()
70        } else {
71            vec![None]
72        };
73        let original = graph.original(part_id);
74        let mut cuts = (0..original.0.len()).map(|i| i as f64).collect::<Vec<_>>();
75        cuts.extend(targets.iter().flat_map(|&(a, b, _)| [a, b]));
76        cuts.sort_by(f64::total_cmp);
77        cuts.dedup();
78        let coords = cuts
79            .iter()
80            .map(|&t| {
81                let i = (t.floor() as usize).min(original.0.len() - 2);
82                crate::engine::matching::point(
83                    geo::Line::new(original.0[i], original.0[i + 1]),
84                    t - i as f64,
85                )
86            })
87            .collect::<Vec<_>>();
88        for destination in destinations {
89            let selected = cuts
90                .windows(2)
91                .map(|pair| {
92                    let covered = targets.iter().any(|&(a, b, r)| {
93                        pair[0] >= a && pair[1] <= b && destination.is_none_or(|d| d == r)
94                    });
95                    if overlap { covered } else { !covered }
96                })
97                .collect::<Vec<_>>();
98            for piece in selected_pieces(&coords, &selected) {
99                if Euclidean.length(&piece) > t.tolerance() {
100                    violations.push(violation(
101                        spec,
102                        if overlap {
103                            ViolationKind::Overlap
104                        } else {
105                            ViolationKind::NotCovered
106                        },
107                        piece,
108                        Some(part.reference),
109                        destination,
110                    ));
111                }
112            }
113        }
114    }
115    Ok(aggregate(violations, Merge::AllDimensions))
116}

118fn uncovered(
119    t: &Topology,
120    spec: RuleSpec,
121    other: LayerId,
122    mode: Boundary,
123) -> Result<Vec<Violation>, Error> {
124    let source = t.parts(spec.layer)?;
125    let target = t.parts(other)?;
126    let source_lines = if mode == Boundary::BoundariesCovered {
127        cross::boundaries(source)
128    } else {
129        source.lines.as_slice()
130    };
131    let mut violations = Vec::new();
132    for source in source_lines {
133        crate::cancel::check();
134        let mut coords = Vec::<Coord>::new();
135        let mut selected = Vec::new();
136        for segment in source.geometry.lines() {
137            let query = index::expanded(segment.bounding_rect(), t.tolerance());
138            let mut covered = Vec::new();
139            let mut cover = |boundary| {
140                covered.extend(proximity::capsule(segment, boundary, t.tolerance()));
141            };
142            if mode == Boundary::BoundariesCovered {
143                let view = cross::segments(target);
144                for item in view.index.locate_in_envelope_intersecting(&query) {
145                    cover(view.segments[item.data].geometry);
146                }
147            } else {
148                let locations = crate::engine::location::locations(target);
149                for item in target
150                    .polygon_index()
151                    .locate_in_envelope_intersecting(&query)
152                {
153                    let location = &locations[item.data];
154                    for edge in location.tree.locate_in_envelope_intersecting(&query) {
155                        cover(location.segments[edge.data]);
156                    }
157                }
158            }
159            let covered = proximity::union(covered);
160            for [start, end] in proximity::outside(Q::zero(), Q::one(), &covered) {
161                append(
162                    &mut coords,
163                    &mut selected,
164                    segment,
165                    start.outward(segment, false),
166                    end.outward(segment, true),
167                    true,
168                );
169            }
170            // Covered intervals keep residual runs separate and preserve source
171            // vertices. No new path is inserted into the original geometry.
172            let start = coords.last().copied().unwrap_or(segment.start);
173            append(
174                &mut coords,
175                &mut selected,
176                segment,
177                start,
178                segment.end,
179                false,
180            );
181        }
182        violations.extend(
183            selected_pieces(&coords, &selected)
184                .into_iter()
185                .map(|piece| {
186                    violation(
187                        spec,
188                        ViolationKind::NotCovered,
189                        piece,
190                        Some(source.reference.at(spec.layer)),
191                        None,
192                    )
193                }),
194        );
195    }
196    Ok(aggregate(violations, Merge::AllDimensions))
197}

199fn append(
200    coords: &mut Vec<Coord>,
201    selected: &mut Vec<bool>,
202    segment: Line,
203    start: Coord,
204    end: Coord,
205    keep: bool,
206) {
207    if coords.is_empty() {
208        coords.push(segment.start);
209    }
210    if coords.last() != Some(&start) {
211        coords.push(start);
212        selected.push(false);
213    }
214    if coords.last() != Some(&end) {
215        coords.push(end);
216        selected.push(keep);
217    }
218}
