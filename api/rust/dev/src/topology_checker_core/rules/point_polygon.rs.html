10/// How each point part relates to the polygon layer.
11#[derive(Debug, Clone, Copy)]
12pub(super) enum PointTest {
13    /// Each point must be inside a polygon or within tolerance of its boundary.
14    Covered,
15    /// No point may be strictly inside a polygon, farther than tolerance from its boundary.
16    NotInside,
17    /// No point may be inside a polygon or within tolerance of its boundary.
18    NotIntersecting,
19}
20/// How many point features each polygon feature must strictly contain.
21#[derive(Debug, Clone, Copy)]
22pub(super) enum PointCount {
23    /// At least one; uncontained points are allowed.
24    AtLeastOne,
25    /// Exactly one, and every point feature belongs to exactly one polygon feature.
26    ExactlyOne,
27}

29/// Test every point part against the polygon parts near it.
30pub(super) fn points(
31    t: &Topology,
32    spec: RuleSpec,
33    other: LayerId,
34    test: PointTest,
35) -> Result<Vec<Violation>, Error> {
36    let a = t.parts(spec.layer)?;
37    let b = t.parts(other)?;
38    let locations = crate::engine::location::locations(b);
39    let violations = a
40        .points
41        .par_iter()
42        .flat_map_iter(|p| {
43            let mut candidates = b
44                .polygon_index()
45                .locate_in_envelope_intersecting(&index::at(p.geometry.0, t.tolerance()))
46                .map(|i| i.data)
47                .collect::<Vec<_>>();
48            candidates.sort_unstable();
49            let hits = candidates
50                .into_iter()
51                .filter(|&i| {
52                    let polygon = &locations[i];
53                    match test {
54                        PointTest::NotInside => {
55                            polygon.position(p.geometry.0) == 1
56                                && !polygon.near(p.geometry.0, t.tolerance())
57                        }
58                        PointTest::Covered | PointTest::NotIntersecting => {
59                            polygon.position(p.geometry.0) >= 0
60                                || polygon.near(p.geometry.0, t.tolerance())
61                        }
62                    }
63                })
64                .collect::<Vec<_>>();
65            let origin = Some(p.reference.at(spec.layer));
66            let kind = match test {
67                PointTest::Covered => {
68                    return if hits.is_empty() {
69                        vec![violation(
70                            spec,
71                            ViolationKind::NotCovered,
72                            p.geometry,
73                            origin,
74                            None,
75                        )]
76                    } else {
77                        Vec::new()
78                    };
79                }
80                PointTest::NotInside => ViolationKind::Contained,
81                PointTest::NotIntersecting => ViolationKind::Intersection,
82            };
83            hits.into_iter()
84                .map(|i| {
85                    violation(
86                        spec,
87                        kind,
88                        p.geometry,
89                        origin,
90                        Some(b.polygons[i].reference.at(other)),
91                    )
92                })
93                .collect()
94        })
95        .collect();
96    Ok(aggregate(violations, Merge::AllDimensions))
97}

99/// Report point parts strictly inside polygon parts, with polygon origin provenance.
100pub(super) fn contained_points(
101    t: &Topology,
102    spec: RuleSpec,
103    other: LayerId,
104) -> Result<Vec<Violation>, Error> {
105    let a = t.parts(spec.layer)?;
106    let b = t.parts(other)?;
107    let locations = crate::engine::location::locations(a);
108    let violations = index::pairs(a.polygon_index(), b.point_index(), t.tolerance(), false)
109        .par_bridge()
110        .filter_map(|(i, j)| {
111            let (p, q) = (&a.polygons[i], &b.points[j]);
112            (locations[i].position(q.geometry.0) == 1
113                && !locations[i].near(q.geometry.0, t.tolerance()))
114            .then(|| {
115                violation(
116                    spec,
117                    ViolationKind::Contained,
118                    q.geometry,
119                    Some(p.reference.at(spec.layer)),
120                    Some(q.reference.at(other)),
121                )
122            })
123        })
124        .collect();
125    Ok(aggregate(violations, Merge::AllDimensions))
126}

128/// Count point features whose retained parts all lie strictly inside a polygon feature.
129pub(super) fn point_counts(
130    t: &Topology,
131    spec: RuleSpec,
132    other: LayerId,
133    count: PointCount,
134) -> Result<Vec<Violation>, Error> {
135    let av = cross::views(t.parts(spec.layer)?);
136    let bv = cross::views(t.parts(other)?);
137    let mut assignments = vec![0usize; bv.features.len()];
138    let mut counts = vec![0; av.features.len()];
139    for (i, j) in index::pairs(&av.feature_index, &bv.feature_index, 0., false) {
140        let (polygon, point) = (&av.features[i], &bv.features[j]);
141        if !point.points.is_empty()
142            && point
143                .points
144                .iter()
145                .all(|p| polygon.position(p.0) == 1 && !polygon.near(p.0, t.tolerance()))
146        {
147            assignments[j] += 1;
148            counts[i] += 1;
149        }
150    }
151    let (kind, valid): (_, fn(usize) -> bool) = match count {
152        PointCount::AtLeastOne => (ViolationKind::MissingPoint, |n| n > 0),
153        PointCount::ExactlyOne => (ViolationKind::PointCount, |n| n == 1),
154    };
155    let mut violations = Vec::new();
156    for (i, p) in av.features.iter().enumerate() {
157        if !valid(counts[i]) {
158            violations.push(violation(
159                spec,
160                kind,
161                super::feature_geometry(t, spec, spec.layer, p)?,
162                Some(p.reference.at(spec.layer)),
163                None,
164            ));
165        }
166    }
167    if matches!(count, PointCount::ExactlyOne) {
168        for (j, p) in bv.features.iter().enumerate() {
169            if assignments[j] != 1 {
170                violations.push(violation(
171                    spec,
172                    ViolationKind::PointCount,
173                    super::feature_geometry(t, spec, other, p)?,
174                    None,
175                    Some(p.reference.at(other)),
176                ));
177            }
178        }
179    }
180    Ok(aggregate(violations, Merge::AllDimensions))
181}
