1//! Required and prohibited proximity of points and line endpoints to other features.
2use super::violation;
3use crate::{
4    Error, LayerId, RuleSpec, Topology, Violation, ViolationKind,
5    aggregate::{Merge, aggregate},
6    engine::{cross, index, proximity},
7    model::{Part, Parts},
8};
9use geo::{Line, Point};
10use rayon::prelude::*;

12/// Secondary-layer features that cover a point within tolerance.
13#[derive(Debug, Clone, Copy)]
14pub(super) enum Target {
15    /// Any segment of the linework.
16    Linework,
17    /// An endpoint of an open line part.
18    LineEndpoints,
19    /// An exterior or hole boundary segment, including shared edges.
20    PolygonBoundaries,
21    /// A point part.
22    Points,
23}
24
25fn on_segments(
26    point: &Point,
27    segments: &[Part<Line>],
28    tree: &index::Index,
29    tolerance: f64,
30) -> bool {
31    tree.locate_in_envelope_intersecting(&index::at(point.0, tolerance))
32        .any(|i| proximity::near_segment(point.0, segments[i.data].geometry, tolerance))
33}
34fn on_points(point: &Point, points: &[Part<Point>], tree: &index::Index, tolerance: f64) -> bool {
35    tree.locate_in_envelope_intersecting(&index::at(point.0, tolerance))
36        .any(|i| proximity::coincident(points[i.data].geometry.0, point.0, tolerance))
37}
38fn covered(point: &Point, target: Target, parts: &Parts, tolerance: f64) -> bool {
39    match target {
40        Target::Linework => {
41            let view = cross::segments(parts);
42            on_segments(point, &view.segments, &view.index, tolerance)
43        }
44        Target::LineEndpoints => {
45            let view = cross::endpoints(parts);
46            view.index
47                .locate_in_envelope_intersecting(&index::at(point.0, tolerance))
48                .any(|i| {
49                    view.is_open(i.data, tolerance)
50                        && proximity::coincident(view.points[i.data].geometry.0, point.0, tolerance)
51                })
52        }
53        Target::PolygonBoundaries => {
54            let locations = crate::engine::location::locations(parts);
55            parts
56                .polygon_index()
57                .locate_in_envelope_intersecting(&index::at(point.0, tolerance))
58                .any(|item| locations[item.data].near(point.0, tolerance))
59        }
60        Target::Points => on_points(point, &parts.points, parts.point_index(), tolerance),
61    }
62}
63fn uncovered(
64    t: &Topology,
65    spec: RuleSpec,
66    other: LayerId,
67    target: Target,
68    sources: &[Part<Point>],
69    endpoints: Option<&cross::Endpoints>,
70) -> Result<Vec<Violation>, Error> {
71    let b = t.parts(other)?;
72    let violations = sources
73        .par_iter()
74        .enumerate()
75        .filter(|(i, p)| {
76            endpoints.is_none_or(|ends| ends.is_open(*i, t.tolerance()))
77                && !covered(&p.geometry, target, b, t.tolerance())
78        })
79        .map(|(_, p)| {
80            violation(
81                spec,
82                ViolationKind::NotCovered,
83                p.geometry,
84                Some(p.reference.at(spec.layer)),
85                None,
86            )
87        })
88        .collect();
89    Ok(aggregate(violations, Merge::AllDimensions))
90}

92/// Report point parts that no target covers.
93pub(super) fn points(
94    t: &Topology,
95    spec: RuleSpec,
96    other: LayerId,
97    target: Target,
98) -> Result<Vec<Violation>, Error> {
99    uncovered(t, spec, other, target, &t.parts(spec.layer)?.points, None)
100}
101/// Report open line part endpoints that no target covers.
102pub(super) fn endpoints(
103    t: &Topology,
104    spec: RuleSpec,
105    other: LayerId,
106    target: Target,
107) -> Result<Vec<Violation>, Error> {
108    let a = cross::endpoints(t.parts(spec.layer)?);
109    uncovered(t, spec, other, target, &a.points, Some(a))
110}
111/// Report each point within tolerance of a line part.
112pub(super) fn line_contacts(
113    t: &Topology,
114    spec: RuleSpec,
115    other: LayerId,
116) -> Result<Vec<Violation>, Error> {
117    let a = t.parts(spec.layer)?;
118    let b = t.parts(other)?;
119    let violations = index::pairs(a.point_index(), b.line_index(), t.tolerance(), false)
120        .par_bridge()
121        .filter_map(|(i, j)| {
122            let (p, l) = (&a.points[i], &b.lines[j]);
123            l.geometry
124                .lines()
125                .any(|line| proximity::near_segment(p.geometry.0, line, t.tolerance()))
126                .then(|| {
127                    violation(
128                        spec,
129                        ViolationKind::Intersection,
130                        p.geometry,
131                        Some(p.reference.at(spec.layer)),
132                        Some(l.reference.at(other)),
133                    )
134                })
135        })
136        .collect();
137    Ok(aggregate(violations, Merge::AllDimensions))
138}
