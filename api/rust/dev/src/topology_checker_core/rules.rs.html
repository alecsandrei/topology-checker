20pub(crate) fn run(t: &Topology, spec: RuleSpec) -> Result<Vec<Violation>, Error> {
21    // Registration guarantees a secondary layer for every binary rule.
22    let other = || {
23        spec.other.ok_or(Error::Inputs {
24            rule: spec.rule,
25            expected: 2,
26        })
27    };
28    match spec.rule {
29        Rule::GeometryMustNotBeMultipart => geometry::multipart(t, spec),
30        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith => point::overlap(t, spec),
31        Rule::PointMustBeInside => point::inside(t, spec, other()?),
32        Rule::PolygonMustNotOverlap | Rule::PolygonMustNotOverlapWith => polygon::overlap(t, spec),
33        Rule::PolygonMustNotHaveGaps => polygon::gaps(t, spec),
34        Rule::LineMustBeInside => polygon::line_inside(t, spec, other()?),
35        Rule::LineMustNotHaveDangles => line::run(t, spec, LineCheck::Dangles),
36        Rule::LineMustNotHavePseudoNodes => line::run(t, spec, LineCheck::PseudoNodes),
37        Rule::LineMustNotOverlap | Rule::LineMustNotOverlapWith => {
38            line::run(t, spec, LineCheck::Overlap)
39        }
40        Rule::LineMustNotSelfOverlap => line::run(t, spec, LineCheck::SelfOverlap),
41        Rule::LineMustNotSelfIntersect => line::run(t, spec, LineCheck::SelfIntersect),
42        Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
43            line::run(t, spec, LineCheck::Intersect { touch: false })
44        }
45        Rule::LineMustNotIntersectOrTouchInterior
46        | Rule::LineMustNotIntersectOrTouchInteriorWith => {
47            line::run(t, spec, LineCheck::Intersect { touch: true })
48        }
49        Rule::PointMustBeCoveredByLines => coverage::points(t, spec, other()?, Target::Linework),
50        Rule::PointMustBeCoveredByLineEndpoints => {
51            coverage::points(t, spec, other()?, Target::LineEndpoints)
52        }
53        Rule::PointMustBeCoveredByPolygonBoundaries => {
54            coverage::points(t, spec, other()?, Target::PolygonBoundaries)
55        }
56        Rule::LineEndpointsMustBeCoveredByPoints => {
57            coverage::endpoints(t, spec, other()?, Target::Points)
58        }
59        Rule::LineEndpointsMustBeCoveredByPolygonBoundaries => {
60            coverage::endpoints(t, spec, other()?, Target::PolygonBoundaries)
61        }
62        Rule::PointMustNotIntersectLines => coverage::line_contacts(t, spec, other()?),
63        Rule::PointMustBeCoveredByPolygons => {
64            point_polygon::points(t, spec, other()?, PointTest::Covered)
65        }
66        Rule::PointMustNotBeInsidePolygons => {
67            point_polygon::points(t, spec, other()?, PointTest::NotInside)
68        }
69        Rule::PointMustNotIntersectPolygons => {
70            point_polygon::points(t, spec, other()?, PointTest::NotIntersecting)
71        }
72        Rule::PolygonMustNotContainPoints => point_polygon::contained_points(t, spec, other()?),
73        Rule::PolygonMustContainPoint => {
74            point_polygon::point_counts(t, spec, other()?, PointCount::AtLeastOne)
75        }
76        Rule::PolygonMustContainOnePoint => {
77            point_polygon::point_counts(t, spec, other()?, PointCount::ExactlyOne)
78        }
79        Rule::GeometryMustNotBeContainedBy => containment::contained(t, spec, other()?),
80        Rule::LineMustBeCoveredByPolygonBoundaries => {
81            boundary::linework(t, spec, other()?, Boundary::LinesCovered)
82        }
83        Rule::PolygonBoundariesMustBeCoveredByLines => {
84            boundary::linework(t, spec, other()?, Boundary::BoundariesCovered)
85        }
86        Rule::LineMustNotOverlapPolygonBoundaries => {
87            boundary::linework(t, spec, other()?, Boundary::NotShared)
88        }
89        Rule::LineMustBeProperlyInsidePolygons => line_polygon::properly_inside(t, spec, other()?),
90        Rule::LineMustNotCrossPolygonBoundaries => {
91            line_polygon::run(t, spec, other()?, LinePolygon::Cross)
92        }
93        Rule::LineMustNotOverlapWithPolygons => {
94            line_polygon::run(t, spec, other()?, LinePolygon::Overlap)
95        }
96        Rule::LineMustNotIntersectPolygons => {
97            line_polygon::run(t, spec, other()?, LinePolygon::Intersect)
98        }
99    }
100}

117/// Extract connected selected pieces before applying the length threshold.
118/// A closed path's first and last runs form one piece across the traversal seam.
119fn selected_pieces(coords: &[Coord], selected: &[bool]) -> Vec<LineString> {
120    selected_runs(coords, selected)
121        .into_iter()
122        .map(|run| LineString(run.into_iter().map(|i| coords[i]).collect()))
123        .collect()
124}
125/// Coordinate indices of each piece of `selected_pieces`.
126fn selected_runs(coords: &[Coord], selected: &[bool]) -> Vec<Vec<usize>> {
127    let mut runs = Vec::new();
128    let mut run = Vec::new();
129    for (i, &keep) in selected.iter().enumerate() {
130        if keep {
131            if run.is_empty() {
132                run.push(i);
133            }
134            run.push(i + 1);
135        } else if !run.is_empty() {
136            runs.push(std::mem::take(&mut run));
137        }
138    }
139    if !run.is_empty() {
140        runs.push(run);
141    }
142    if runs.len() > 1
143        && coords.first() == coords.last()
144        && selected.first() == Some(&true)
145        && selected.last() == Some(&true)
146    {
147        let first = runs.remove(0);
148        let last = runs.last_mut().expect("two runs before joining");
149        last.extend(first.into_iter().skip(1));
150    }
151    runs
152}

154/// Emit deterministic aggregated violations, with a bounded duplicate-pair path.
155pub(crate) fn run_into<E>(
156    t: &Topology,
157    spec: RuleSpec,
158    mut sink: impl FnMut(Violation) -> Result<(), E>,
159) -> Result<(), crate::SinkError<E>> {
160    if matches!(
161        spec.rule,
162        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith
163    ) {
164        point::overlap_into(t, spec, sink)
165    } else if matches!(
166        spec.rule,
167        Rule::PolygonMustNotHaveGaps
168            | Rule::GeometryMustNotBeMultipart
169            | Rule::PolygonMustContainPoint
170            | Rule::PolygonMustContainOnePoint
171            | Rule::LineMustNotHaveDangles
172            | Rule::LineMustNotHavePseudoNodes
173    ) {
174        emit(run(t, spec)?, &mut sink)
175    } else {
176        // Geometry scratch belongs to one original feature and its candidate
177        // targets. No complete feature-pair report is retained.
178        let mut snapshot = t.snapshot();
179        let mut fids = t
180            .layer(spec.layer)?
181            .features()
182            .map(|(fid, _)| fid)
183            .collect::<Vec<_>>();
184        fids.sort_unstable();
185        let kinds: &[ViolationKind] = match spec.rule {
186            Rule::LineMustNotSelfIntersect => {
187                &[ViolationKind::SelfOverlap, ViolationKind::SelfCrossing]
188            }
189            Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
190                &[ViolationKind::Overlap, ViolationKind::Crossing]
191            }
192            Rule::LineMustNotIntersectOrTouchInterior
193            | Rule::LineMustNotIntersectOrTouchInteriorWith => &[
194                ViolationKind::Overlap,
195                ViolationKind::Crossing,
196                ViolationKind::TouchInterior,
197            ],
198            _ => &[],
199        };
200        for kind in if kinds.is_empty() {
201            vec![None]
202        } else {
203            kinds.iter().copied().map(Some).collect()
204        } {
205            for &fid in &fids {
206                crate::cancel::check();
207                snapshot.select_feature(spec.layer, fid)?;
208                let mut violations = run(&snapshot, spec)?;
209                if let Some(kind) = kind {
210                    violations.retain(|v| v.kind == kind);
211                }
212                emit(violations, &mut sink)?;
213            }
214        }
215        Ok(())
216    }
217}
218fn emit<E>(
219    mut violations: Vec<Violation>,
220    sink: &mut impl FnMut(Violation) -> Result<(), E>,
221) -> Result<(), crate::SinkError<E>> {
222    violations.sort_by_key(|v| {
223        (
224            v.kind,
225            v.origin.map(FeatureRef::feature),
226            v.destination.map(FeatureRef::feature),
227        )
228    });
229    for v in violations {
230        crate::cancel::check();
231        sink(v).map_err(crate::SinkError::Consumer)?;
232    }
233    Ok(())
234}

236fn feature_geometry(
237    t: &Topology,
238    spec: RuleSpec,
239    layer: crate::LayerId,
240    feature: &crate::engine::cross::Feature,
241) -> Result<Geometry, Error> {
242    t.cancellable(|| feature.output()).map_err(|e| {
243        e.precision_context(
244            "polygon feature emission",
245            spec.rule,
246            format!(
247                ", layer {}, FID {}",
248                t.layer(layer).expect("registered layer").name(),
249                feature.reference.fid
250            ),
251        )
252    })
