20pub(crate) fn run(t: &Topology, spec: RuleSpec) -> Result<Vec<Violation>, Error> {
21    // Registration guarantees a secondary layer for every binary rule.
22    let other = || {
23        spec.other.ok_or(Error::Inputs {
24            rule: spec.rule,
25            expected: 2,
26        })
27    };
28    match spec.rule {
29        Rule::GeometryMustNotBeMultipart => geometry::multipart(t, spec),
30        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith => point::overlap(t, spec),
31        Rule::PointMustBeInside => point::inside(t, spec, other()?),
32        Rule::PolygonMustNotOverlap | Rule::PolygonMustNotOverlapWith => polygon::overlap(t, spec),
33        Rule::PolygonMustNotHaveGaps => polygon::gaps(t, spec),
34        Rule::LineMustBeInside => polygon::line_inside(t, spec, other()?),
35        Rule::LineMustNotHaveDangles => line::run(t, spec, LineCheck::Dangles),
36        Rule::LineMustNotHavePseudoNodes => line::run(t, spec, LineCheck::PseudoNodes),
37        Rule::LineMustNotOverlap | Rule::LineMustNotOverlapWith => {
38            line::run(t, spec, LineCheck::Overlap)
39        }
40        Rule::LineMustNotSelfOverlap => line::run_self(t, spec, false),
41        Rule::LineMustNotSelfIntersect => line::run_self(t, spec, true),
42        Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
43            line::run(t, spec, LineCheck::Intersect { touch: false })
44        }
45        Rule::LineMustNotIntersectOrTouchInterior
46        | Rule::LineMustNotIntersectOrTouchInteriorWith => {
47            line::run(t, spec, LineCheck::Intersect { touch: true })
48        }
49        Rule::PointMustBeCoveredByLines => coverage::points(t, spec, other()?, Target::Linework),
50        Rule::PointMustBeCoveredByLineEndpoints => {
51            coverage::points(t, spec, other()?, Target::LineEndpoints)
52        }
53        Rule::PointMustBeCoveredByPolygonBoundaries => {
54            coverage::points(t, spec, other()?, Target::PolygonBoundaries)
55        }
56        Rule::LineEndpointsMustBeCoveredByPoints => {
57            coverage::endpoints(t, spec, other()?, Target::Points)
58        }
59        Rule::LineEndpointsMustBeCoveredByPolygonBoundaries => {
60            coverage::endpoints(t, spec, other()?, Target::PolygonBoundaries)
61        }
62        Rule::PointMustNotIntersectLines => coverage::line_contacts(t, spec, other()?),
63        Rule::PointMustBeCoveredByPolygons => {
64            point_polygon::points(t, spec, other()?, PointTest::Covered)
65        }
66        Rule::PointMustNotBeInsidePolygons => {
67            point_polygon::points(t, spec, other()?, PointTest::NotInside)
68        }
69        Rule::PointMustNotIntersectPolygons => {
70            point_polygon::points(t, spec, other()?, PointTest::NotIntersecting)
71        }
72        Rule::PolygonMustNotContainPoints => point_polygon::contained_points(t, spec, other()?),
73        Rule::PolygonMustContainPoint => {
74            point_polygon::point_counts(t, spec, other()?, PointCount::AtLeastOne)
75        }
76        Rule::PolygonMustContainOnePoint => {
77            point_polygon::point_counts(t, spec, other()?, PointCount::ExactlyOne)
78        }
79        Rule::GeometryMustNotBeContainedBy => containment::contained(t, spec, other()?),
80        Rule::LineMustBeCoveredByPolygonBoundaries => {
81            boundary::uncovered(t, spec, other()?, Boundary::LinesCovered)
82        }
83        Rule::PolygonBoundariesMustBeCoveredByLines => {
84            boundary::uncovered(t, spec, other()?, Boundary::BoundariesCovered)
85        }
86        Rule::LineMustNotOverlapPolygonBoundaries => boundary::shared(t, spec, other()?),
87        Rule::LineMustBeProperlyInsidePolygons => line_polygon::properly_inside(t, spec, other()?),
88        Rule::LineMustNotCrossPolygonBoundaries => {
89            line_polygon::run(t, spec, other()?, LinePolygon::Cross)
90        }
91        Rule::LineMustNotOverlapWithPolygons => {
92            line_polygon::run(t, spec, other()?, LinePolygon::Overlap)
93        }
94        Rule::LineMustNotIntersectPolygons => {
95            line_polygon::run(t, spec, other()?, LinePolygon::Intersect)
96        }
97    }
98}

100fn violation(
101    spec: RuleSpec,
102    kind: ViolationKind,
103    geometry: impl Into<Geometry>,
104    origin: Option<FeatureRef>,

115/// Extract connected selected pieces before applying the length threshold.
116/// A closed path's first and last runs form one piece across the traversal seam.
117fn selected_pieces(coords: &[Coord], selected: &[bool]) -> Vec<LineString> {
118    selected_runs(coords, selected)
119        .into_iter()
120        .map(|run| LineString(run.into_iter().map(|i| coords[i]).collect()))
121        .collect()
122}
123/// Coordinate indices of each piece of `selected_pieces`.
124fn selected_runs(coords: &[Coord], selected: &[bool]) -> Vec<Vec<usize>> {
125    let mut runs = Vec::new();
126    let mut run = Vec::new();
127    for (i, &keep) in selected.iter().enumerate() {
128        if keep {
129            if run.is_empty() {
130                run.push(i);
131            }
132            run.push(i + 1);
133        } else if !run.is_empty() {
134            runs.push(std::mem::take(&mut run));
135        }
136    }
137    if !run.is_empty() {
138        runs.push(run);
139    }
140    if runs.len() > 1
141        && coords.first() == coords.last()
142        && selected.first() == Some(&true)
143        && selected.last() == Some(&true)
144    {
145        let first = runs.remove(0);
146        let last = runs.last_mut().expect("two runs before joining");
147        last.extend(first.into_iter().skip(1));
148    }
149    runs
150}

152/// Emit deterministic aggregated violations, with a bounded duplicate-pair path.
153pub(crate) fn run_into<E>(
154    t: &Topology,
155    spec: RuleSpec,
156    mut sink: impl FnMut(Violation) -> Result<(), E>,
157) -> Result<(), crate::SinkError<E>> {
158    if matches!(
159        spec.rule,
160        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith
161    ) {
162        point::overlap_into(t, spec, sink)
163    } else if matches!(
164        spec.rule,
165        Rule::PolygonMustNotHaveGaps
166            | Rule::GeometryMustNotBeMultipart
167            | Rule::PolygonMustContainPoint
168            | Rule::PolygonMustContainOnePoint
169            | Rule::LineMustNotHaveDangles
170            | Rule::LineMustNotHavePseudoNodes
171    ) {
172        emit(run(t, spec)?, &mut sink)
173    } else {
174        // Geometry scratch belongs to one original feature and its candidate
175        // targets. No complete feature-pair report is retained.
176        let mut snapshot = t.snapshot();
177        let mut fids = t
178            .layer(spec.layer)?
179            .features()
180            .map(|(fid, _)| fid)
181            .collect::<Vec<_>>();
182        fids.sort_unstable();
183        let kinds: &[ViolationKind] = match spec.rule {
184            Rule::LineMustNotSelfIntersect => {
185                &[ViolationKind::SelfOverlap, ViolationKind::SelfCrossing]
186            }
187            Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
188                &[ViolationKind::Overlap, ViolationKind::Crossing]
189            }
190            Rule::LineMustNotIntersectOrTouchInterior
191            | Rule::LineMustNotIntersectOrTouchInteriorWith => &[
192                ViolationKind::Overlap,
193                ViolationKind::Crossing,
194                ViolationKind::TouchInterior,
195            ],
196            _ => &[],
197        };
198        for kind in if kinds.is_empty() {
199            vec![None]
200        } else {
201            kinds.iter().copied().map(Some).collect()
202        } {
203            for &fid in &fids {
204                crate::cancel::check();
205                snapshot.select_feature(spec.layer, fid)?;
206                let mut violations = run(&snapshot, spec)?;
207                if let Some(kind) = kind {
208                    violations.retain(|v| v.kind == kind);
209                }
210                emit(violations, &mut sink)?;
211            }
212        }
213        Ok(())
214    }
215}
216fn emit<E>(
217    mut violations: Vec<Violation>,
218    sink: &mut impl FnMut(Violation) -> Result<(), E>,
219) -> Result<(), crate::SinkError<E>> {
220    violations.sort_by_key(|v| {
221        (
222            v.kind,
223            v.origin.map(FeatureRef::feature),
224            v.destination.map(FeatureRef::feature),
225        )
226    });
227    for v in violations {
228        crate::cancel::check();
229        sink(v).map_err(crate::SinkError::Consumer)?;
230    }
231    Ok(())
232}

234fn feature_geometry(
235    t: &Topology,
236    spec: RuleSpec,
237    layer: crate::LayerId,
238    feature: &crate::engine::cross::Feature,
239) -> Result<Geometry, Error> {
240    t.cancellable(|| feature.output()).map_err(|e| {
241        e.precision_context(
242            "polygon feature emission",
243            spec.rule,
244            format!(
245                ", layer {}, FID {}",
246                t.layer(layer).expect("registered layer").name(),
247                feature.reference.fid
248            ),
249        )
250    })
251}

253#[cfg(test)]
254mod tests {
255    use super::*;
256    use crate::{Layer, model::PartRef};
257    use geo::{GeometryCollection, LineString, Polygon, Rect, coord};
258
259    #[test]
260    fn feature_emission_failures_name_their_layer_and_feature() {
261        // The union of these overlapping slivers cannot be rounded.
262        let x = 1f64.next_up();
263        let rect = Rect::new(coord! {x: 1., y: 0.}, coord! {x: x, y: 1.}).to_polygon();
264        let triangle = Polygon::new(
265            LineString::from(vec![(1., -1.), (x, -1.), (1., 1.), (1., -1.)]),
266            vec![],
267        );
268        let area = crate::engine::arrangement::overlay([&rect, &triangle], [], false);
269        let feature = crate::engine::cross::Feature {
270            reference: PartRef { fid: 5, part: None },
271            geometry: Geometry::GeometryCollection(GeometryCollection(vec![])),
272            points: Vec::new(),
273            locations: Vec::new(),
274            area: Some(area),
275            non_areal: Vec::new(),
276        };
277        let mut topology = Topology::new(0.).unwrap();
278        let polygons = topology.add_layer(Layer::from_geometries("parcels", []));
279        let points = topology.add_layer(Layer::from_geometries("wells", []));
280        let spec = topology
281            .add_rule(Rule::PolygonMustContainPoint, polygons, Some(points))
282            .unwrap();
283        let error = feature_geometry(&topology, spec, polygons, &feature).unwrap_err();
284        assert!(
285            matches!(
286                &error,
287                Error::Precision { operation, rule: Some(Rule::PolygonMustContainPoint), context, .. }
288                    if operation == "polygon feature emission" && context == ", layer parcels, FID 5"
289            ),
290            "{error:?}"
291        );
292    }
