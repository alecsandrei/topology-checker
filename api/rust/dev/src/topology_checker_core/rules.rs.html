31pub(crate) fn run(t: &Topology, spec: RuleSpec) -> Result<Vec<Violation>, Error> {
32    // A worker thread can pick this rule up inside another validation's scope.
33    t.cancellable(|| dispatch(t, spec))

35fn dispatch(t: &Topology, spec: RuleSpec) -> Result<Vec<Violation>, Error> {
36    // Registration guarantees a secondary layer for every binary rule.
37    let other = || {
38        spec.other.ok_or(Error::Inputs {
39            rule: spec.rule,
40            expected: 2,
41        })
42    };
43    match spec.rule {
44        Rule::GeometryMustNotBeMultipart => geometry::multipart(t, spec),
45        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith => point::overlap(t, spec),
46        Rule::PointMustBeInside => point::inside(t, spec, other()?),
47        Rule::PolygonMustNotOverlap | Rule::PolygonMustNotOverlapWith => polygon::overlap(t, spec),
48        Rule::PolygonMustNotHaveGaps => polygon::gaps(t, spec),
49        Rule::LineMustBeInside => polygon::line_inside(t, spec, other()?),
50        Rule::LineMustNotHaveDangles => line::run(t, spec, LineCheck::Dangles),
51        Rule::LineMustNotHavePseudoNodes => line::run(t, spec, LineCheck::PseudoNodes),
52        Rule::LineMustNotOverlap | Rule::LineMustNotOverlapWith => {
53            line::run(t, spec, LineCheck::Overlap)
54        }
55        Rule::LineMustNotSelfOverlap => line::run_self(t, spec, false),
56        Rule::LineMustNotSelfIntersect => line::run_self(t, spec, true),
57        Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
58            line::run(t, spec, LineCheck::Intersect { touch: false })
59        }
60        Rule::LineMustNotIntersectOrTouchInterior
61        | Rule::LineMustNotIntersectOrTouchInteriorWith => {
62            line::run(t, spec, LineCheck::Intersect { touch: true })
63        }
64        Rule::PointMustBeCoveredByLines => coverage::points(t, spec, other()?, Target::Linework),
65        Rule::PointMustBeCoveredByLineEndpoints => {
66            coverage::points(t, spec, other()?, Target::LineEndpoints)
67        }
68        Rule::PointMustBeCoveredByPolygonBoundaries => {
69            coverage::points(t, spec, other()?, Target::PolygonBoundaries)
70        }
71        Rule::LineEndpointsMustBeCoveredByPoints => {
72            coverage::endpoints(t, spec, other()?, Target::Points)
73        }
74        Rule::LineEndpointsMustBeCoveredByPolygonBoundaries => {
75            coverage::endpoints(t, spec, other()?, Target::PolygonBoundaries)
76        }
77        Rule::PointMustNotIntersectLines => coverage::line_contacts(t, spec, other()?),
78        Rule::PointMustBeCoveredByPolygons => {
79            point_polygon::points(t, spec, other()?, PointTest::Covered)
80        }
81        Rule::PointMustNotBeInsidePolygons => {
82            point_polygon::points(t, spec, other()?, PointTest::NotInside)
83        }
84        Rule::PointMustNotIntersectPolygons => {
85            point_polygon::points(t, spec, other()?, PointTest::NotIntersecting)
86        }
87        Rule::PolygonMustNotContainPoints => point_polygon::contained_points(t, spec, other()?),
88        Rule::PolygonMustContainPoint => {
89            point_polygon::point_counts(t, spec, other()?, PointCount::AtLeastOne)
90        }
91        Rule::PolygonMustContainOnePoint => {
92            point_polygon::point_counts(t, spec, other()?, PointCount::ExactlyOne)
93        }
94        Rule::GeometryMustNotBeContainedBy => containment::contained(t, spec, other()?),
95        Rule::LineMustBeCoveredByPolygonBoundaries => {
96            boundary::uncovered(t, spec, other()?, Boundary::LinesCovered)
97        }
98        Rule::PolygonBoundariesMustBeCoveredByLines => {
99            boundary::uncovered(t, spec, other()?, Boundary::BoundariesCovered)
100        }
101        Rule::LineMustNotOverlapPolygonBoundaries => boundary::shared(t, spec, other()?),
102        Rule::LineMustBeProperlyInsidePolygons => line_polygon::properly_inside(t, spec, other()?),
103        Rule::LineMustNotCrossPolygonBoundaries => {
104            line_polygon::run(t, spec, other()?, LinePolygon::Cross)
105        }
106        Rule::LineMustNotOverlapWithPolygons => {
107            line_polygon::run(t, spec, other()?, LinePolygon::Overlap)
108        }
109        Rule::LineMustNotIntersectPolygons => {
110            line_polygon::run(t, spec, other()?, LinePolygon::Intersect)
111        }
112    }
113}

130/// Extract connected selected pieces before applying the length threshold.
131/// A closed path's first and last runs form one piece across the traversal seam.
132fn selected_pieces(coords: &[Coord], selected: &[bool]) -> Vec<LineString> {
133    selected_runs(coords, selected)
134        .into_iter()
135        .map(|run| LineString(run.into_iter().map(|i| coords[i]).collect()))
136        .collect()
137}
138/// Coordinate indices of each piece of `selected_pieces`.
139fn selected_runs(coords: &[Coord], selected: &[bool]) -> Vec<Vec<usize>> {
140    let mut runs = Vec::new();
141    let mut run = Vec::new();
142    for (i, &keep) in selected.iter().enumerate() {
143        if keep {
144            if run.is_empty() {
145                run.push(i);
146            }
147            run.push(i + 1);
148        } else if !run.is_empty() {
149            runs.push(std::mem::take(&mut run));
150        }
151    }
152    if !run.is_empty() {
153        runs.push(run);
154    }
155    if runs.len() > 1
156        && coords.first() == coords.last()
157        && selected.first() == Some(&true)
158        && selected.last() == Some(&true)
159    {
160        let first = runs.remove(0);
161        let last = runs.last_mut().expect("two runs before joining");
162        last.extend(first.into_iter().skip(1));
163    }
164    runs
165}

167/// Emit deterministic aggregated violations, with a bounded duplicate-pair path.
168pub(crate) fn run_into<E>(
169    t: &Topology,
170    spec: RuleSpec,
171    mut sink: impl FnMut(Violation) -> Result<(), E>,
172) -> Result<(), crate::SinkError<E>> {
173    if matches!(
174        spec.rule,
175        Rule::PointMustNotOverlap | Rule::PointMustNotOverlapWith
176    ) {
177        point::overlap_into(t, spec, sink)
178    } else if matches!(
179        spec.rule,
180        Rule::PolygonMustNotHaveGaps
181            | Rule::GeometryMustNotBeMultipart
182            | Rule::PolygonMustContainPoint
183            | Rule::PolygonMustContainOnePoint
184            | Rule::LineMustNotHaveDangles
185            | Rule::LineMustNotHavePseudoNodes
186    ) {
187        emit(run(t, spec)?, &mut sink)
188    } else {
189        per_feature_into(t, spec, FEATURES_PER_THREAD, &mut sink)
190    }
191}
192/// Features per thread in one parallel batch of [`per_feature_into`]. Output waits
193/// for a whole batch, so batches stay small enough to stream early and to bound the
194/// violations held at once, and large enough to balance uneven features.
195const FEATURES_PER_THREAD: usize = 8;
196/// Most features per thread in a batch of [`per_feature_into`] that grew after
197/// [quick](QUICK_PER_HELPER) ones.
198const MOST_PER_THREAD: usize = 256;
199/// A parallel batch of [`per_feature_into`] that takes less than this per helper
200/// thread doubles the features per thread of the next one, so that starting its
201/// helpers, some 20 µs each, stays a small part of it. One that takes four times as
202/// long starts over.
203const QUICK_PER_HELPER: Duration = Duration::from_micros(500);
204/// A batch of [`per_feature_into`] that takes at least this long hands the next one to
205/// a thread apart from the calling thread, which holds the caches the first feature
206/// built. The scratch of slow features is faster to reach there, which outweighs the
207/// 10 µs or so each handover costs.
208const SLOW_BATCH: Duration = Duration::from_millis(1);
209/// Helper threads in use by the feature batches of every validation in the process.
210static HELPERS: AtomicUsize = AtomicUsize::new(0);
211/// Helper threads of one feature batch, returned when it ends.
212struct Helpers {
213    threads: usize,
214    shared: usize,
215    budget: &'static AtomicUsize,
216}
217impl Helpers {
218    /// Up to `want` of the helpers free now. All validations together use one helper
219    /// per CPU but one, and a batch never waits for them: without any it runs on the
220    /// calling thread alone.
221    fn take(want: usize) -> Self {
222        #[cfg(test)]
223        if let Some(free) = tests::free() {
224            return Self {
225                threads: want.min(free),
226                shared: 0,
227                budget: &HELPERS,
228            };
229        }
230        // Tests that fix the thread count get it whatever runs beside them.
231        #[cfg(test)]
232        if tests::threads().is_some() || Workers::forced_here().is_some() {
233            return Self {
234                threads: want,
235                shared: 0,
236                budget: &HELPERS,
237            };
238        }
239        static LIMIT: OnceLock<usize> = OnceLock::new();
240        let limit = *LIMIT.get_or_init(|| {
241            std::thread::available_parallelism().map_or(1, std::num::NonZero::get) - 1
242        });
243        Self::take_from(&HELPERS, limit, want)
244    }
245    /// Up to `want` of `limit` helpers, `budget` counting those in use.
246    fn take_from(budget: &'static AtomicUsize, limit: usize, want: usize) -> Self {
247        let mut used = budget.load(atomic::Ordering::Relaxed);
248        loop {
249            let threads = want.min(limit.saturating_sub(used));
250            if threads == 0 {
251                return Self {
252                    threads,
253                    shared: 0,
254                    budget,
255                };
256            }
257            match budget.compare_exchange_weak(
258                used,
259                used + threads,
260                atomic::Ordering::Relaxed,
261                atomic::Ordering::Relaxed,
262            ) {
263                Ok(_) => {
264                    return Self {
265                        threads,
266                        shared: threads,
267                        budget,
268                    };
269                }
270                Err(now) => used = now,
271            }
272        }
273    }
274}
275impl Drop for Helpers {
276    fn drop(&mut self) {
277        self.budget
278            .fetch_sub(self.shared, atomic::Ordering::Relaxed);
279    }
280}

284pub(super) fn in_feature_task() -> bool {

287/// `task` of each item, results in order. A feature task runs them on the threads it
288/// holds, other callers on rayon.
289pub(super) fn each<I: Send, R: Send>(items: Vec<I>, task: impl Fn(I) -> R + Sync + Send) -> Vec<R> {
290    if in_feature_task() {
291        Workers::all().each(items, task)
292    } else {
293        items.into_par_iter().map(task).collect()
294    }
295}
296/// The items of `task(i)` for each `i` in `0..len`, in order. Runs like [`each`], in
297/// ranges of indices.
298pub(super) fn flat_map<I: IntoIterator<Item: Send>>(
299    len: usize,
300    task: impl Fn(usize) -> I + Sync + Send,
301) -> Vec<I::Item> {
302    if in_feature_task() {
303        Workers::all()
304            .chunks(len, |range| range.flat_map(&task).collect::<Vec<_>>())
305            .into_iter()
306            .flatten()
307            .collect()
308    } else {
309        (0..len).into_par_iter().flat_map_iter(task).collect()
310    }
311}
312/// Candidate pairs held at once.
313const PAIRS: usize = 1 << 14;
314/// Candidate pairs a thread of a feature task takes at a time.
315const PAIR_TASK: usize = 64;
316/// The results `task` returns for candidate pairs, in pair order, run like [`each`]
317/// on [`PAIRS`] pairs at a time. The threads of a feature task take [`PAIR_TASK`]
318/// pairs at a time. On rayon the next pairs are found meanwhile, as threads taking
319/// pairs one at a time from the search wait for each other.
320pub(super) fn each_pair<R: Send>(
321    mut pairs: impl Iterator<Item = (usize, usize)> + Send,
322    task: impl Fn((usize, usize)) -> Option<R> + Sync + Send,
323) -> Vec<R> {
324    let mut results = Vec::new();
325    let mut next = || pairs.by_ref().take(PAIRS).collect::<Vec<_>>();
326    if in_feature_task() {
327        let workers = Workers::all();
328        loop {
329            let chunk = next();
330            if chunk.is_empty() {
331                return results;
332            }
333            if chunk.len() <= PAIR_TASK || workers.alone() {
334                results.extend(chunk.into_iter().filter_map(&task));
335            } else {
336                results.extend(pair_tasks(workers, &chunk, &task).into_iter().flatten());
337            }
338        }
339    }
340    let mut chunk = next();
341    while !chunk.is_empty() {
342        let found;
343        (found, chunk) = rayon::join(
344            || {
345                chunk
346                    .par_iter()
347                    .filter_map(|&pair| task(pair))
348                    .collect::<Vec<_>>()
349            },
350            &mut next,
351        );
352        results.extend(found);
353    }
354    results
355}
356/// The results of `task` for `pairs`, [`PAIR_TASK`] pairs to a task, in pair order.
357fn pair_tasks<R: Send>(
358    workers: Workers,
359    pairs: &[(usize, usize)],
360    task: &(impl Fn((usize, usize)) -> Option<R> + Sync),
361) -> Vec<Vec<R>> {
362    workers.run(pairs.len().div_ceil(PAIR_TASK), |k| {
363        let pairs = pairs.iter().skip(k * PAIR_TASK).take(PAIR_TASK);
364        pairs.filter_map(|&pair| task(pair)).collect()
365    })
366}
367/// Kinds of a per-feature rule in emission order. Empty when each feature emits all
368/// of its violations together.
369fn feature_kinds(rule: Rule) -> &'static [ViolationKind] {
370    match rule {
371        Rule::LineMustNotSelfIntersect => {
372            &[ViolationKind::SelfOverlap, ViolationKind::SelfCrossing]
373        }
374        Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
375            &[ViolationKind::Overlap, ViolationKind::Crossing]
376        }
377        Rule::LineMustNotIntersectOrTouchInterior
378        | Rule::LineMustNotIntersectOrTouchInteriorWith => &[
379            ViolationKind::Overlap,
380            ViolationKind::Crossing,
381            ViolationKind::TouchInterior,
382        ],
383        _ => &[],
384    }
385}
386/// Run a rule on one original feature at a time and emit in kind order, then feature
387/// order. Geometry scratch belongs to one feature and its candidate targets, and no
388/// complete feature-pair report is retained. Batches of `per_thread` features per
389/// thread run in parallel and each feature runs once: its first kind is emitted with
390/// its batch, and later kinds wait until every feature has emitted the first. Caches
391/// every feature shares, such as the line graph, build on every thread of the
392/// validation, whichever feature needs them first and whatever helpers are free. The
393/// first feature runs alone, so the others need not wait for them. Features of a batch
394/// smaller than its threads split their own steps over the spare ones, and threads out
395/// of features join the steps of those still running. Batches run on the calling
396/// thread, or after a [slow](SLOW_BATCH) one on a thread apart, and on helpers of their
397/// own, never on a shared pool, so cancelling stops them as soon as their features
398/// check, whatever else runs in the process.
399fn per_feature_into<E>(
400    t: &Topology,
401    spec: RuleSpec,
402    per_thread: usize,
403    sink: &mut impl FnMut(Violation) -> Result<(), E>,
404) -> Result<(), crate::SinkError<E>> {
405    let mut fids = t
406        .layer(spec.layer)?
407        .features()
408        .map(|(fid, _)| fid)
409        .collect::<Vec<_>>();
410    fids.sort_unstable();
411    let kinds = feature_kinds(spec.rule);
412    // Holding later kinds saves a rerun per kind; a million held added about 10% to peak.
413    let mut later = kinds.iter().skip(1).map(|_| Vec::new()).collect::<Vec<_>>();
414    let all = Workers::all().threads();
415    #[cfg(test)]
416    let all = tests::threads().unwrap_or(all).max(1);
417    let feature = |snapshot: &mut Topology, fid: i64, threads: usize| {
418        // The selection checks this topology's token on whichever thread runs it.
419        t.cancellable(|| {
420            parallel::task(threads, all, || {
421                #[cfg(test)]
422                tests::visit(t, spec, fid)?;
423                snapshot.select_feature(spec.layer, fid)?;
424                let mut violations = run(snapshot, spec)?;
425                sort(&mut violations);
426                Ok::<_, Error>(violations)
427            })
428        })
429    };
430    // A snapshot per thread, as cheap features slow down sharing its making.
431    let batch = |(fids, threads, steps): (&[i64], usize, usize)| {
432        Workers::with_threads(threads).run_sharing(
433            fids.len(),
434            steps,
435            || t.snapshot(),
436            |snapshot, i| feature(snapshot, fids[i], steps),
437        )
438    };
439    #[cfg(not(test))]
440    let (slow, quick) = (SLOW_BATCH, QUICK_PER_HELPER);
441    #[cfg(test)]
442    let (slow, quick) = (tests::slow_batch(), tests::quick_per_helper());
443    parallel::apart(batch, |apart| {
444        let mut rest = &fids[..];
445        let (mut first, mut after_slow) = (true, false);
446        let mut grown = per_thread.max(1);
447        while !rest.is_empty() {
448            crate::cancel::check();
449            let helpers = Helpers::take(all - 1);
450            let threads = 1 + helpers.threads;
451            let size = if first { 1 } else { threads * grown };
452            let (fids, tail) = rest.split_at(size.min(rest.len()));
453            // Fewer features than threads split their own steps over the rest.
454            let steps = (threads / fids.len()).max(1);
455            let started = Instant::now();
456            let results = if after_slow {
457                apart((fids, threads, steps))
458            } else {
459                batch((fids, threads, steps))
460            };
461            let took = started.elapsed();
462            after_slow = took >= slow;
463            if !first && helpers.threads > 0 {
464                let helpers = u32::try_from(helpers.threads).unwrap_or(u32::MAX);
465                grown = grow(grown, per_thread, took, quick.saturating_mul(helpers));
466            }
467            (rest, first) = (tail, false);
468            drop(helpers);
469            for violations in results {
470                for v in violations? {
471                    match kinds.iter().position(|&kind| kind == v.kind) {
472                        Some(0) => send(v, sink)?,
473                        Some(i) => later[i - 1].push(v),
474                        None if kinds.is_empty() => send(v, sink)?,
475                        None => {}
476                    }
477                }
478            }
479        }
480        Ok::<_, crate::SinkError<E>>(())
481    })?;
482    later.into_iter().flatten().try_for_each(|v| send(v, sink))
483}
484/// Features per thread for the batch after one of `per` that took `took`: twice as
485/// many after one [quicker](QUICK_PER_HELPER) than `quick`, and `least` again after
486/// one four times as slow.
487fn grow(per: usize, least: usize, took: Duration, quick: Duration) -> usize {
488    let least = least.max(1);
489    if took < quick {
490        per.saturating_mul(2).min(MOST_PER_THREAD.max(least))
491    } else if took >= quick.saturating_mul(4) {
492        least
493    } else {
494        per
495    }
496}
497fn emit<E>(
498    mut violations: Vec<Violation>,
499    sink: &mut impl FnMut(Violation) -> Result<(), E>,
500) -> Result<(), crate::SinkError<E>> {
501    sort(&mut violations);
502    violations.into_iter().try_for_each(|v| send(v, sink))
503}
504/// Stable sort into emission order. Cancelling unwinds with every violation still
505/// in the slice.
506fn sort(violations: &mut [Violation]) {
507    let key = |v: &Violation| {
508        (
509            v.kind,
510            v.origin.map(FeatureRef::feature),
511            v.destination.map(FeatureRef::feature),
512        )
513    };
514    let mut compared = 0;
515    violations.sort_by(|a, b| {
516        crate::cancel::every(compared, "violations.sort");
517        compared += 1;
518        key(a).cmp(&key(b))
519    });
520}
521fn send<E>(
522    v: Violation,
523    sink: &mut impl FnMut(Violation) -> Result<(), E>,
524) -> Result<(), crate::SinkError<E>> {
525    crate::cancel::check();
526    sink(v).map_err(crate::SinkError::Consumer)
527}

635    /// Longest wait for another thread in a test.
636    const PATIENCE: Duration = Duration::from_secs(60);
637    /// Threads that reached a point, and whether they may pass it.
638    #[derive(Debug, Default)]
639    struct Gate {
640        state: Mutex<Passage>,
641        changed: Condvar,
642    }
643    #[derive(Debug, Default)]
644    struct Passage {
645        arrivals: usize,
646        threads: HashSet<ThreadId>,
647        open: bool,
648    }
649    impl Gate {
650        fn state(&self) -> MutexGuard<'_, Passage> {
651            self.state.lock().unwrap_or_else(PoisonError::into_inner)
652        }
653        /// Count this thread in and return how many arrived before it.
654        fn arrive(&self) -> usize {
655            let mut state = self.state();
656            state.arrivals += 1;
657            state.threads.insert(std::thread::current().id());
658            self.changed.notify_all();
659            state.arrivals - 1
660        }
661        fn open(&self) {
662            self.state().open = true;
663            self.changed.notify_all();
664        }
665        /// Block until `ready` holds, for at most [`PATIENCE`].
666        fn wait(&self, ready: impl Fn(&Passage) -> bool) -> MutexGuard<'_, Passage> {
667            self.wait_for(PATIENCE, ready)
668        }
669        /// Block until `ready` holds, for at most `timeout`.
670        fn wait_for(
671            &self,
672            timeout: Duration,
673            ready: impl Fn(&Passage) -> bool,
674        ) -> MutexGuard<'_, Passage> {
675            self.changed
676                .wait_timeout_while(self.state(), timeout, |state| !ready(state))
677                .unwrap_or_else(PoisonError::into_inner)
678                .0
679        }
680    }

683    enum Action {
684        Cancel(CancellationToken),
685        Fail,
686        Panic(&'static str),
687        Hold(Duration),
688        Note(Arc<Mutex<Vec<Instant>>>),
689        /// Note the feature and the thread it runs on.
690        Thread(Arc<Mutex<Vec<(i64, ThreadId)>>>),
691        /// Arrive at the gate and go on, but wait until it opens when this many
692        /// arrived before.
693        Arrive(Arc<Gate>, usize),
694        /// Arrive at the gate and wait until it opens, then check for cancellation as
695        /// rule code does and, if that does not unwind, sleep before returning.
696        Block(Arc<Gate>, Duration),
697        /// Wait until this many threads arrived at the gate.
698        Await(Arc<Gate>, usize),
699    }
700    /// How long a feature blocked at a gate keeps running when cancelling misses it.
701    const GRACE: Duration = Duration::from_secs(1);
702    /// Actions keyed by primary layer name and FID, taken as a worker starts that feature.
703    static ACTIONS: Mutex<Vec<(String, i64, Action)>> = Mutex::new(Vec::new());
704    pub(super) fn visit(t: &Topology, spec: RuleSpec, fid: i64) -> Result<(), Error> {
705        let Ok(layer) = t.layer(spec.layer) else {
706            return Ok(());
707        };
708        let action = ACTIONS
709            .lock()
710            .unwrap_or_else(PoisonError::into_inner)
711            .iter()
712            .find(|(name, at, _)| name == layer.name() && *at == fid)
713            .map(|(_, _, action)| action.clone());
714        match action {
715            Some(Action::Cancel(token)) => token.cancel(),
716            Some(Action::Fail) => {
717                return Err(Error::Precision {
718                    operation: "feature".into(),
719                    rule: Some(spec.rule),
720                    context: format!(", FID {fid}"),
721                    reason: "injected".into(),
722                });
723            }
724            Some(Action::Panic(message)) => std::panic::panic_any(message),
725            Some(Action::Hold(duration)) => std::thread::sleep(duration),
726            Some(Action::Note(starts)) => starts
727                .lock()
728                .unwrap_or_else(PoisonError::into_inner)
729                .push(Instant::now()),
730            Some(Action::Thread(threads)) => threads
731                .lock()
732                .unwrap_or_else(PoisonError::into_inner)
733                .push((fid, std::thread::current().id())),
734            Some(Action::Arrive(gate, wait)) => {
735                let earlier = gate.arrive();
736                if earlier == wait {
737                    drop(gate.wait(|state| state.open));
738                }
739            }
740            Some(Action::Block(gate, then)) => {
741                gate.arrive();
742                drop(gate.wait(|state| state.open));
743                crate::cancel::check();
744                std::thread::sleep(then);
745            }
746            Some(Action::Await(gate, arrivals)) => {
747                drop(gate.wait(|state| state.arrivals >= arrivals));
748            }
749            None => {}
750        }
751        Ok(())
752    }
753    fn on(t: &Topology, spec: RuleSpec, fid: i64, action: Action) -> String {
754        let name = t.layer(spec.layer).unwrap().name().to_string();
755        ACTIONS
756            .lock()
757            .unwrap_or_else(PoisonError::into_inner)
758            .push((name.clone(), fid, action));
759        name
760    }
761    fn clear(name: &str) {
762        ACTIONS
763            .lock()
764            .unwrap_or_else(PoisonError::into_inner)
765            .retain(|(at, ..)| at != name);
766    }

782    /// Emitted violations, then the result.
783    type Emission = (Vec<Violation>, Result<(), Error>);
784    /// Emission by a sequential pass per kind, each rerunning every feature in FID
785    /// order on one reused snapshot.
786    fn reference_emission(t: &Topology, spec: RuleSpec) -> Emission {
787        if !per_feature(spec.rule) {
788            return emission(t, spec, None);
789        }
790        let mut out = Vec::new();
791        let result = (|| {
792            let mut snapshot = t.snapshot();
793            let mut fids = t
794                .layer(spec.layer)?
795                .features()
796                .map(|(fid, _)| fid)
797                .collect::<Vec<_>>();
798            fids.sort_unstable();
799            let kinds: &[ViolationKind] = match spec.rule {
800                Rule::LineMustNotSelfIntersect => {
801                    &[ViolationKind::SelfOverlap, ViolationKind::SelfCrossing]
802                }
803                Rule::LineMustNotIntersect | Rule::LineMustNotIntersectWith => {
804                    &[ViolationKind::Overlap, ViolationKind::Crossing]
805                }
806                Rule::LineMustNotIntersectOrTouchInterior
807                | Rule::LineMustNotIntersectOrTouchInteriorWith => &[
808                    ViolationKind::Overlap,
809                    ViolationKind::Crossing,
810                    ViolationKind::TouchInterior,
811                ],
812                _ => &[],
813            };
814            for kind in if kinds.is_empty() {
815                vec![None]
816            } else {
817                kinds.iter().copied().map(Some).collect()
818            } {
819                for &fid in &fids {
820                    visit(t, spec, fid)?;
821                    snapshot.select_feature(spec.layer, fid)?;
822                    let mut violations = run(&snapshot, spec)?;
823                    if let Some(kind) = kind {
824                        violations.retain(|v| v.kind == kind);
825                    }
826                    sort(&mut violations);
827                    out.extend(violations);
828                }
829            }
830            Ok(())
831        })();
832        (out, result)
833    }
834    /// Emission by [`run_into`], or by [`per_feature_into`] with a given batch size.
835    fn emission(t: &Topology, spec: RuleSpec, batch: Option<usize>) -> Emission {
836        let mut out = Vec::new();
837        let mut sink = |v| {
838            out.push(v);
839            Ok::<_, Infallible>(())
840        };
841        let result = match batch {
842            Some(batch) => per_feature_into(t, spec, batch, &mut sink),
843            None => run_into(t, spec, &mut sink),
844        };
845        (out, result.map_err(Error::from))
846    }
847    fn reference(t: &Topology, spec: RuleSpec) -> Result<Vec<Violation>, Error> {
848        let (out, result) = reference_emission(t, spec);
849        result.map(|()| out)
850    }

864    /// Seeded xorshift, so a failure reproduces from its seed.
865    struct Rng(u64);
866    impl Rng {
867        fn new(seed: u64) -> Self {
868            Self((seed + 1).wrapping_mul(0x9E37_79B9_7F4A_7C15))
869        }
870        fn next(&mut self) -> u64 {
871            self.0 ^= self.0 << 13;
872            self.0 ^= self.0 >> 7;
873            self.0 ^= self.0 << 17;
874            self.0
875        }
876        fn below(&mut self, n: usize) -> usize {
877            (self.next() % n as u64) as usize
878        }
879        fn chance(&mut self, percent: usize) -> bool {
880            self.below(100) < percent
881        }
882        fn pick<T: Copy>(&mut self, items: &[T]) -> T {
883            items[self.below(items.len())]
884        }
885    }
886    /// Coordinates `offset + scale * (grid + jitter)` on a 7 by 7 grid, so features
887    /// often share vertices, segments and near misses.
888    #[derive(Clone, Copy)]
889    struct Frame {
890        offset: f64,
891        scale: f64,
892        jitter: f64,
893    }
894    impl Frame {
895        fn random(rng: &mut Rng) -> Self {
896            Self {
897                offset: rng.pick(&[0., -250., 1e3, 1e6, 1e9]),
898                scale: rng.pick(&[1., 0.01, 100.]),
899                jitter: rng.pick(&[0., 0., 1e-4, 0.3]),
900            }
901        }
902        fn tolerance(self, rng: &mut Rng) -> f64 {
903            self.scale * rng.pick(&[0., 0., 1e-3, 0.05, 0.6, 1.5])
904        }
905        fn at(self, rng: &mut Rng, x: usize, y: usize) -> Coord {
906            let mut shift = || {
907                if self.jitter == 0. || rng.chance(50) {
908                    0.
909                } else {
910                    self.jitter * (rng.below(3) as f64 - 1.)
911                }
912            };
913            coord! {
914                x: self.offset + self.scale * (x as f64 + shift()),
915                y: self.offset + self.scale * (y as f64 + shift()),
916            }
917        }
918        fn point(self, rng: &mut Rng) -> Coord {
919            let (x, y) = (rng.below(7), rng.below(7));
920            self.at(rng, x, y)
921        }
922    }
923    /// A part that may continue from `from`. Some parts are a single coordinate, zero
924    /// length or closed.
925    fn line_part(rng: &mut Rng, frame: Frame, from: Option<Coord>) -> LineString {
926        let start = match from {
927            Some(from) if rng.chance(60) => from,
928            _ => frame.point(rng),
929        };
930        let mut coords = vec![start];
931        match rng.below(14) {
932            0 => {}
933            1 => coords.push(start),
934            _ => {
935                for _ in 0..1 + rng.below(4) {
936                    let last = *coords.last().unwrap();
937                    let next = frame.point(rng);
938                    coords.push(match rng.below(3) {
939                        0 => coord! { x: last.x, y: next.y },
940                        1 => coord! { x: next.x, y: last.y },
941                        _ => next,
942                    });
943                }
944                if coords.len() > 2 && rng.chance(20) {
945                    coords.push(start);
946                }
947            }
948        }
949        LineString(coords)
950    }
951    fn line(rng: &mut Rng, frame: Frame, from: Option<Coord>) -> Geometry {
952        let mut parts = vec![line_part(rng, frame, from)];
953        while parts.len() < 4 && rng.chance(30) {
954            let end = parts.last().and_then(|part| part.0.last().copied());
955            parts.push(line_part(rng, frame, end));
956        }
957        if parts.len() == 1 && rng.chance(75) {
958            parts.remove(0).into()
959        } else {
960            MultiLineString(parts).into()
961        }
962    }
963    fn polygon(rng: &mut Rng, frame: Frame) -> Polygon {
964        let (x, y) = (rng.below(6), rng.below(6));
965        let (w, h) = (1 + rng.below(6 - x), 1 + rng.below(6 - y));
966        let mut ring = |corners: &[(usize, usize)]| {
967            let mut ring = corners
968                .iter()
969                .map(|&(i, j)| frame.at(rng, i, j))
970                .collect::<Vec<_>>();
971            ring.push(ring[0]);
972            LineString(ring)
973        };
974        let exterior = if w == h && (x + y) % 3 == 0 {
975            ring(&[(x, y), (x + w, y), (x, y + h)])
976        } else {
977            ring(&[(x, y), (x + w, y), (x + w, y + h), (x, y + h)])
978        };
979        let holes = if w > 2 && h > 2 && (x + y) % 2 == 0 {
980            let (u, v) = (x + w - 1, y + h - 1);
981            vec![ring(&[(x + 1, y + 1), (u, y + 1), (u, v), (x + 1, v)])]
982        } else {
983            Vec::new()
984        };
985        Polygon::new(exterior, holes)
986    }
987    fn feature(rng: &mut Rng, frame: Frame, kind: GeometryKind, from: Option<Coord>) -> Geometry {
988        match kind {
989            GeometryKind::Line => line(rng, frame, from),
990            GeometryKind::Point if rng.chance(70) => Point(frame.point(rng)).into(),
991            GeometryKind::Point => MultiPoint(
992                (0..1 + rng.below(3))
993                    .map(|_| Point(frame.point(rng)))
994                    .collect(),
995            )
996            .into(),
997            GeometryKind::Polygon if rng.chance(70) => polygon(rng, frame).into(),
998            GeometryKind::Polygon => {
999                MultiPolygon((0..1 + rng.below(2)).map(|_| polygon(rng, frame)).collect()).into()
1000            }
1001            _ => match rng.below(4) {
1002                0 => feature(rng, frame, GeometryKind::Line, from),
1003                1 => feature(rng, frame, GeometryKind::Point, from),
1004                2 => feature(rng, frame, GeometryKind::Polygon, from),
1005                _ => Geometry::GeometryCollection(GeometryCollection(vec![
1006                    Point(frame.point(rng)).into(),
1007                    line(rng, frame, from),
1008                ])),
1009            },
1010        }
1011    }

1019    /// Up to `size` features in shuffled FID order. Some duplicate or reverse an earlier
1020    /// feature or one of `shared`, and lines often continue from the previous feature.
1021    fn layer(
1022        rng: &mut Rng,
1023        frame: Frame,
1024        name: String,
1025        kind: GeometryKind,
1026        size: usize,
1027        shared: &[Geometry],
1028    ) -> Layer {
1029        let n = rng.below(size + 1);
1030        let mut fids = (0..n as i64).map(|i| i * 3 - 7).collect::<Vec<_>>();
1031        for i in (1..fids.len()).rev() {
1032            fids.swap(i, rng.below(i + 1));
1033        }
1034        let mut geometries: Vec<Geometry> = Vec::new();
1035        for _ in 0..n {
1036            let copies = geometries.len() + shared.len();
1037            let geometry = if copies > 0 && rng.chance(15) {
1038                let i = rng.below(copies);
1039                let mut copy = geometries
1040                    .get(i)
1041                    .unwrap_or_else(|| &shared[i - geometries.len()])
1042                    .clone();
1043                if let Geometry::LineString(line) = &mut copy
1044                    && rng.chance(50)
1045                {
1046                    line.0.reverse();
1047                }
1048                copy
1049            } else {
1050                let from = geometries.last().and_then(end);
1051                feature(rng, frame, kind, from)
1052            };
1053            geometries.push(geometry);
1054        }
1055        Layer::from_features(name, fids.into_iter().zip(geometries))
1056    }

1061    /// A seeded primary and secondary layer of every family, and each rule that
1062    /// registers on them. Only valid layers that a rule uses are added.
1063    fn case(seed: u64, name: &str, size: usize, rules: &[Rule]) -> Case {
1064        let mut rng = Rng::new(seed);
1065        let frame = Frame::random(&mut rng);
1066        let mut topology = Topology::new(frame.tolerance(&mut rng)).unwrap();
1067        let kinds = [
1068            GeometryKind::Line,
1069            GeometryKind::Point,
1070            GeometryKind::Polygon,
1071            GeometryKind::Any,
1072        ];
1073        let mut slots = Vec::new();
1074        for kind in kinds {
1075            let a = layer(
1076                &mut rng,
1077                frame,
1078                format!("{name}-{kind:?}-a"),
1079                kind,
1080                size,
1081                &[],
1082            );
1083            let shared = a.features().map(|(_, g)| g.clone()).collect::<Vec<_>>();
1084            let b = layer(
1085                &mut rng,
1086                frame,
1087                format!("{name}-{kind:?}-b"),
1088                kind,
1089                size,
1090                &shared,
1091            );
1092            slots.push((Some(a), None));
1093            slots.push((Some(b), None));
1094        }
1095        let slot = |kind, side| 2 * kinds.iter().position(|&k| k == kind).unwrap() + side;
1096        let mut add = |topology: &mut Topology, i: usize| {
1097            let (layer, id) = &mut slots[i];
1098            if let Some(layer) = layer.take() {
1099                *id = layer.parts().is_ok().then(|| topology.add_layer(layer));
1100            }
1101            *id
1102        };
1103        let mut specs = Vec::new();
1104        for &rule in rules {
1105            let (a, b) = rule.inputs();
1106            let primary = add(&mut topology, slot(a, 0));
1107            let other = b.map(|b| add(&mut topology, slot(b, 1)));
1108            if let (Some(primary), None | Some(Some(_))) = (primary, other)
1109                && let Ok(spec) = topology.add_rule(rule, primary, other.flatten())
1110            {
1111                specs.push(spec);
1112            }
1113        }
1114        Case { topology, specs }
1115    }

1143    /// Compare new emission with the reference for every spec of one case, on cold
1144    /// caches and on every batch size.
1145    fn differential(seed: u64, size: usize, rules: &[Rule], stats: &mut Stats) {
1146        let cold = case(seed, "diff", size, rules);
1147        let warm = case(seed, "diff", size, rules);
1148        stats.cases += 1;
1149        for (i, (&spec, &cold_spec)) in warm.specs.iter().zip(&cold.specs).enumerate() {
1150            let new = format!("{:?}", emission(&cold.topology, cold_spec, None));
1151            let (expected, result) = reference_emission(&warm.topology, spec);
1152            let debug = format!("{:?}", (&expected, &result));
1153            assert_eq!(new, debug, "seed {seed}, {}", spec.rule);
1154            if per_feature(spec.rule) {
1155                // Batches of one to three features, so most cases span several.
1156                let batch = 1 + (seed as usize + i) % 3;
1157                let batched = format!("{:?}", emission(&warm.topology, spec, Some(batch)));
1158                assert_eq!(batched, debug, "seed {seed}, {}, batch {batch}", spec.rule);
1159            }
1160            if matches!(
1161                spec.rule,
1162                Rule::LineMustNotSelfOverlap | Rule::LineMustNotSelfIntersect
1163            ) {
1164                // Whole-layer self checks keep layer order across parallel features.
1165                let mut snapshot = warm.topology.snapshot();
1166                let sequential = warm
1167                    .topology
1168                    .layer(spec.layer)
1169                    .unwrap()
1170                    .features()
1171                    .map(|(fid, _)| {
1172                        snapshot.select_feature(spec.layer, fid)?;
1173                        run(&snapshot, spec)
1174                    })
1175                    .collect::<Result<Vec<_>, Error>>()
1176                    .map(|features| features.concat());
1177                assert_eq!(
1178                    format!("{:?}", eager(&warm.topology, spec)),
1179                    format!("{sequential:?}"),
1180                    "seed {seed}, {}",
1181                    spec.rule
1182                );
1183            }
1184            stats.specs += 1;
1185            match result.map(|()| expected) {
1186                Ok(violations) => {
1187                    stats.violations += violations.len();
1188                    let first = violations.first().map(|v| v.kind);
