13/// A registration index scoped to the topology that returned it.
14/// IDs from another topology must not be reused, even if their indices match.
15#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
16#[serde(transparent)]

18impl LayerId {

20    #[must_use]
21    pub fn index(self) -> usize {
22        self.0
23    }

26/// Feature provenance within a topology, including an optional original part index.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
28#[non_exhaustive]
29pub struct FeatureRef {
30    /// Topology-scoped layer ID.
31    pub layer: LayerId,
32    /// Signed feature ID, unique within its layer.
33    pub fid: i64,
34    /// Zero-based depth-first part index, including skipped degenerate parts.
35    /// `None` means multiple parts contributed or no single part is identified.
36    pub part: Option<u32>,
37}
38impl FeatureRef {
39    /// Create provenance using an identity from the relevant topology.
40    #[must_use]
41    pub fn new(layer: LayerId, fid: i64, part: Option<u32>) -> Self {
42        Self { layer, fid, part }
43    }
44
45    pub(crate) fn feature(self) -> (LayerId, i64) {
46        (self.layer, self.fid)
47    }
48    pub(crate) fn aggregate(&mut self, other: Self) {
49        if self.part != other.part {
50            self.part = None;
51        }
52    }
53}

55/// Layer-local part provenance, cached independently of any topology.
56///
57/// It deliberately has no equality or ordering: parts from different layers may
58/// share FIDs, so callers attach a `LayerId` with [`PartRef::at`] before comparing
59/// references that are not known to come from one layer.
60#[derive(Debug, Clone, Copy)]
61pub(crate) struct PartRef {
62    pub fid: i64,
63    pub part: Option<u32>,
64}
65impl PartRef {
66    pub(crate) fn at(self, layer: LayerId) -> FeatureRef {
67        FeatureRef {
68            layer,
69            fid: self.fid,
70            part: self.part,
71        }
72    }
73}

75/// A cached layer validation failure, independent of the layer's registration ID.
76#[derive(Debug, Clone, PartialEq, Eq)]
77pub(crate) enum Invalid {
78    DuplicateFid(i64),
79    NonFinite(i64),
80    Polygon {
81        fid: i64,
82        part: u32,
83        ring: Option<usize>,
84        reason: &'static str,
85    },
86}
87impl Invalid {
88    /// The feature that failed validation.
89    pub(crate) fn fid(&self) -> i64 {
90        let (Self::DuplicateFid(fid) | Self::NonFinite(fid) | Self::Polygon { fid, .. }) = self;
91        *fid
92    }
93    pub(crate) fn at(self, layer: LayerId, name: &str) -> Error {
94        let name = name.to_string();
95        match self {
96            Self::DuplicateFid(fid) => Error::DuplicateFid { layer, name, fid },
97            Self::NonFinite(fid) => Error::NonFinite { layer, name, fid },
98            Self::Polygon {
99                fid,
100                part,
101                ring,
102                reason,
103            } => Error::InvalidPolygon {
104                layer,
105                name,
106                fid,
107                part,
108                ring,
109                reason: reason.into(),
110            },
111        }
112    }
113}

115/// Broad input geometry families used to check rule compatibility.
116#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
117#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
118#[serde(rename_all = "lowercase")]
119#[non_exhaustive]
120pub enum GeometryKind {
121    /// Points and multipoints.
122    Point,
123    /// Segments, line strings and multiline strings.
124    Line,
125    /// Polygons, multipolygons, rectangles and triangles.
126    Polygon,
127    /// Geometry collections, whose children are checked recursively for rules.
128    Collection,
129    /// Any geometry family, accepted by the multipart and containment rules.
130    Any,
131}
132impl GeometryKind {
133    /// Classify a geometry without validating its coordinates or collection children.
134    #[must_use]
135    pub fn of(g: &Geometry) -> Self {
136        match g {
137            Geometry::Point(_) | Geometry::MultiPoint(_) => Self::Point,
138            Geometry::Line(_) | Geometry::LineString(_) | Geometry::MultiLineString(_) => {
139                Self::Line
140            }
141            Geometry::Polygon(_)
142            | Geometry::MultiPolygon(_)
143            | Geometry::Rect(_)
144            | Geometry::Triangle(_) => Self::Polygon,
145            Geometry::GeometryCollection(_) => Self::Collection,
146        }
147    }
148    /// Return whether geometry of this family can be a rule input requiring `required`.
149    /// Collections are classified by their children, so they never satisfy a family.
150    #[must_use]
151    pub fn satisfies(self, required: Self) -> bool {
152        required == Self::Any || self == required
153    }
154}

156/// An owned planar layer. Cached views cannot be invalidated through mutation.
157///
158/// Caches do not depend on a topology or `LayerId`, so one `Arc<Layer>` can be
159/// registered in several topologies and its cleaned parts, indexes, polygon
160/// union and cross-geometry views are built once and reused.
161///
162/// Construction defers finite-coordinate and duplicate-FID checks until rule
163/// registration, warnings or validation. Cleaning removes consecutive duplicate
164/// coordinates and skips lines with fewer than two coordinates and polygon rings
165/// with fewer than four coordinates or three distinct vertices. A degenerate
166/// exterior skips the polygon; degenerate holes are skipped individually.
167/// Part indices retain the original traversal positions. Coordinate units are
168/// supplied by the caller; the core does not interpret or transform a CRS.
169#[derive(Debug)]
170pub struct Layer {
171    name: String,
172    fids: Vec<i64>,
173    geometries: Arc<Vec<Geometry>>,
174    ranks: Cache<std::collections::BTreeMap<i64, usize>>,
175    parts: Cache<Result<Parts, Invalid>>,
176}
177impl Layer {
178    /// Own geometries in input order, assigning signed FIDs starting at zero.
179    /// Validation is deferred until this layer's parts are requested.
180    pub fn from_geometries(
181        name: impl Into<String>,
182        geometries: impl IntoIterator<Item = Geometry>,
183    ) -> Self {
184        Self::from_features(
185            name,
186            geometries
187                .into_iter()
188                .enumerate()
189                .map(|(i, g)| (i as i64, g)),
190        )
191    }
192    /// Own features in input order, preserving signed FIDs and original geometries.
193    /// FIDs must be unique within this layer; validation is deferred.
194    pub fn from_features(
195        name: impl Into<String>,
196        features: impl IntoIterator<Item = (i64, Geometry)>,
197    ) -> Self {
198        let (fids, geometries) = features.into_iter().unzip();
199        Self {
200            name: name.into(),
201            fids,
202            geometries: Arc::new(geometries),
203            ranks: Cache::new(),
204            parts: Cache::new(),
205        }
206    }
207    /// Return the caller-supplied layer name, which need not be unique.
208    #[must_use]
209    pub fn name(&self) -> &str {
210        &self.name
211    }
212    /// Iterate original, uncleaned features in input order.
213    pub fn features(&self) -> impl ExactSizeIterator<Item = (i64, &Geometry)> {
214        self.fids.iter().copied().zip(self.geometries.iter())
215    }
216    pub(crate) fn ranks(&self) -> &std::collections::BTreeMap<i64, usize> {
217        self.ranks.get_or_init(|| {
218            self.fids
219                .iter()
220                .copied()
221                .enumerate()
222                .map(|(i, fid)| (fid, i))
223                .collect()
224        })
225    }
226    /// Return the original feature at a zero-based input position.
227    #[must_use]
228    pub fn feature(&self, index: usize) -> Option<(i64, &Geometry)> {
229        Some((*self.fids.get(index)?, self.geometries.get(index)?))
230    }
231    /// Return the number of original features, including degenerate geometries.
232    #[must_use]
233    pub fn len(&self) -> usize {
234        self.fids.len()
235    }
236    /// Return whether the layer contains no original features.
237    #[must_use]
238    pub fn is_empty(&self) -> bool {
239        self.fids.is_empty()
240    }
241    pub(crate) fn check_kind(&self, id: LayerId, expected: GeometryKind) -> Result<(), Error> {
242        fn check(g: &Geometry, expected: GeometryKind) -> Option<GeometryKind> {
243            if expected == GeometryKind::Any {
244                return None;
245            }
246            if let Geometry::GeometryCollection(c) = g {
247                return c.iter().find_map(|g| check(g, expected));
248            }
249            let found = GeometryKind::of(g);
250            (!found.satisfies(expected)).then_some(found)
251        }
252        for (fid, g) in self.features() {
253            if let Some(found) = check(g, expected) {
254                return Err(Error::GeometryKind {
255                    layer: id,
256                    name: self.name.clone(),
257                    fid,
258                    expected,
259                    found,
260                });
261            }
262        }
263        Ok(())
264    }
265    /// Validate cleaned geometry without registering a rule.
266    ///
267    /// # Errors
268    /// Returns a deferred input error with the caller's layer identity.
269    pub fn check(&self, id: LayerId) -> Result<(), Error> {
270        self.parts().map(|_| ()).map_err(|e| e.at(id, self.name()))
271    }
272    pub(crate) fn parts(&self) -> Result<&Parts, Invalid> {
273        self.parts
274            .get_or_init(|| self.make_parts())
275            .as_ref()
276            .map_err(Clone::clone)
277    }
278    fn make_parts(&self) -> Result<Parts, Invalid> {
279        let mut out = Parts {
280            source: Some(Arc::clone(&self.geometries)),
281            ..Parts::default()
282        };
283        let mut seen = HashSet::new();
284        for (feature, (fid, g)) in self.features().enumerate() {
285            if !seen.insert(fid) {
286                return Err(Invalid::DuplicateFid(fid));
287            }
288            if g.coords_iter()
289                .any(|c| !c.x.is_finite() || !c.y.is_finite())
290            {
291                return Err(Invalid::NonFinite(fid));
292            }
293            let first = out.polygons.len();
294            out.add(
295                g,
296                PartRef { fid, part: Some(0) },
297                &mut 0,
298                Some(feature),
299                &mut Vec::new(),
300            );
301            for p in &out.polygons[first..] {
302                if let Some((ring, reason)) = crate::engine::validity::polygon(&p.geometry) {
303                    return Err(Invalid::Polygon {
304                        fid,
305                        part: p.reference.part.unwrap_or(0),
306                        ring: Some(p.rings[ring]),
307                        reason,
308                    });
309                }
310            }
311            if let Some(invalid) = out.invalid.take() {
312                return Err(invalid);
313            }
314        }
315        Ok(out)
316    }
317}

343fn clean_line(line: &LineString) -> LineString {
344    let mut coords = line.0.clone();
345    coords.dedup();
346    LineString(coords)
347}
348fn ring_valid(ring: &LineString) -> bool {
349    if ring.0.len() < 4 {
350        return false;
351    }
352    let first = ring.0[0];
353    let Some(second) = ring.0.iter().find(|&&c| c != first) else {
354        return false;
355    };
356    ring.0.iter().any(|c| *c != first && c != second)
357}

359impl Parts {
360    fn feature_ranges(&self) -> &std::collections::BTreeMap<i64, [std::ops::Range<usize>; 3]> {
361        self.ranges.get_or_init(|| {
362            let mut map = std::collections::BTreeMap::new();
363            fn add<G>(
364                map: &mut std::collections::BTreeMap<i64, [std::ops::Range<usize>; 3]>,
365                parts: &[Part<G>],
366                family: usize,
367            ) {
368                for (i, p) in parts.iter().enumerate() {
369                    let ranges = map
370                        .entry(p.reference.fid)
371                        .or_insert_with(|| [0..0, 0..0, 0..0]);
372                    let range = &mut ranges[family];
373                    if range.start == range.end {
374                        range.start = i;
375                    }
376                    range.end = i + 1;
377                }
378            }
379            add(&mut map, &self.points, 0);
380            add(&mut map, &self.lines, 1);
381            add(&mut map, &self.polygons, 2);
382            map
383        })
384    }
385    pub fn point_feature(&self, fid: i64) -> &[Part<Point>] {
386        self.feature_ranges()
387            .get(&fid)
388            .map_or(&[], |r| &self.points[r[0].clone()])
389    }
390    pub fn feature(&self, fid: i64) -> Self {
391        let ranges = self.feature_ranges();
392        let empty = [0..0, 0..0, 0..0];
393        let r = ranges.get(&fid).unwrap_or(&empty);
394        Self {
395            source: self.source.clone(),
396            points: self.points[r[0].clone()].to_vec(),
397            lines: self.lines[r[1].clone()].to_vec(),
398            polygons: self.polygons[r[2].clone()].to_vec(),
399            ..Self::default()
400        }
401    }
402    pub fn point_index(&self) -> &Index {
403        self.point_index.get_or_init(|| {
404            index::build(
405                self.points
406                    .iter()
407                    .enumerate()
408                    .map(|(i, p)| (i, Rect::new(p.geometry.0, p.geometry.0))),
409            )
410        })
411    }
412    pub fn line_index(&self) -> &Index {
413        self.line_index.get_or_init(|| {
414            index::build(
415                self.lines
416                    .iter()
417                    .enumerate()
418                    .filter_map(|(i, p)| p.geometry.bounding_rect().map(|r| (i, r))),
419            )
420        })
421    }
422    pub fn polygon_index(&self) -> &Index {
423        self.polygon_index.get_or_init(|| {
424            index::build(
425                self.polygons
426                    .iter()
427                    .enumerate()
428                    .filter_map(|(i, p)| p.geometry.bounding_rect().map(|r| (i, r))),
429            )
430        })
431    }

437    }

439    fn add_point(&mut self, p: &Point, reference: PartRef, part: &mut u32) {
440        self.points.push(Part {
441            reference: Self::next_reference(reference, part),
442            geometry: *p,
443            rings: Vec::new(),
444        });
445    }
446
447    fn add_line(
448        &mut self,
449        l: &LineString,
450        reference: PartRef,
451        part: &mut u32,
452        feature: Option<usize>,
453        path: &[usize],
454    ) {
455        let reference = Self::next_reference(reference, part);
456        let geometry = match feature {
457            Some(feature) if !l.0.windows(2).any(|p| p[0] == p[1]) => Shared::Source {
458                geometries: Arc::clone(self.source.as_ref().expect("layer storage")),
459                feature,
460                path: path.to_vec(),
461            },
462            _ => Shared::Cleaned(clean_line(l)),
463        };
464        if geometry.0.len() < 2 {
465            self.skipped += 1;
466        } else {
467            self.lines.push(Part {
468                reference,
469                geometry,
470                rings: Vec::new(),
471            });
472        }
473    }
474
475    fn add_polygon(
476        &mut self,
477        p: &Polygon,
478        reference: PartRef,
479        part: &mut u32,
480        feature: Option<usize>,
481        path: &[usize],
482    ) {
483        let reference = Self::next_reference(reference, part);
484        if let Some(feature) = feature {
485            let rings = std::iter::once(p.exterior())
486                .chain(p.interiors())
487                .collect::<Vec<_>>();
488            if rings
489                .iter()
490                .all(|r| ring_valid(r) && !r.0.windows(2).any(|p| p[0] == p[1]))
491            {
492                self.polygons.push(Part {
493                    reference,
494                    geometry: Shared::Source {
495                        geometries: Arc::clone(self.source.as_ref().expect("layer storage")),
496                        feature,
497                        path: path.to_vec(),
498                    },
499                    rings: (0..rings.len()).collect(),
500                });
501                return;
502            }
503        }
504        let exterior = clean_line(p.exterior());
505        if !ring_valid(&exterior) {
506            self.skipped += 1;
507            return;
508        }
509        let mut rings = vec![0];
510        let holes = p
511            .interiors()
512            .iter()
513            .enumerate()
514            .map(|(i, r)| (i + 1, clean_line(r)))
515            .filter_map(|(i, r)| {
516                let valid = ring_valid(&r);
517                if !valid {
518                    self.skipped += 1;
519                }
520                if valid {
521                    rings.push(i);
522                    Some(r)
523                } else {
524                    None
525                }
526            })
527            .collect();
528        self.polygons.push(Part {
529            reference,
530            geometry: Shared::Cleaned(Polygon::new(exterior, holes)),
531            rings,
532        });
533    }
534
535    fn add(
536        &mut self,
537        g: &Geometry,
538        reference: PartRef,
539        part: &mut u32,
540        feature: Option<usize>,
541        path: &mut Vec<usize>,
542    ) {
543        match g {
544            Geometry::MultiPoint(m) => {
545                for p in m {
546                    self.add_point(p, reference, part);
547                }
548            }
549            Geometry::MultiLineString(m) => {
550                for (i, p) in m.into_iter().enumerate() {
551                    path.push(i);
552                    self.add_line(p, reference, part, feature, path);
553                    path.pop();
554                }
555            }
556            Geometry::MultiPolygon(m) => {
557                let first = self.polygons.len();
558                for (i, p) in m.into_iter().enumerate() {
559                    path.push(i);
560                    self.add_polygon(p, reference, part, feature, path);
561                    path.pop();
562                }
563                let polygons = &self.polygons[first..];
564                let locations = polygons
565                    .iter()
566                    .map(|p| crate::engine::location::Location::new(&p.geometry))
567                    .collect::<Vec<_>>();
568                let tree = index::build(
569                    polygons
570                        .iter()
571                        .enumerate()
572                        .filter_map(|(i, p)| p.geometry.bounding_rect().map(|r| (i, r))),
573                );
574                for (i, p) in polygons.iter().enumerate() {
575                    for item in tree.locate_in_envelope_intersecting(&index::envelope(
576                        p.geometry.bounding_rect().expect("clean polygon"),
577                    )) {
578                        if item.data > i
579                            && crate::engine::validity::multipart(
580                                &locations[i],
581                                &locations[item.data],
582                            )
583                        {
584                            self.invalid = Some(Invalid::Polygon {
585                                fid: reference.fid,
586                                part: p.reference.part.unwrap_or(0),
587                                ring: None,
588                                reason: "invalid multipart: nested shells, overlapping parts or shared edges",
589                            });
590                        }
591                    }
592                }
593            }
594            Geometry::GeometryCollection(m) => {
595                for (i, p) in m.into_iter().enumerate() {
596                    path.push(i);
597                    self.add(p, reference, part, feature, path);
598                    path.pop();
599                }
600            }
601            Geometry::Line(l) => self.add_line(
602                &LineString(vec![l.start, l.end]),
603                reference,
604                part,
605                None,
606                path,
607            ),
608            Geometry::Rect(r) => self.add_polygon(&r.to_polygon(), reference, part, None, path),
609            Geometry::Triangle(t) => self.add_polygon(&t.to_polygon(), reference, part, None, path),
610            Geometry::Point(p) => self.add_point(p, reference, part),
611            Geometry::LineString(l) => self.add_line(l, reference, part, feature, path),
612            Geometry::Polygon(p) => self.add_polygon(p, reference, part, feature, path),
613        }
614    }
615}
616pub(crate) fn part_count(g: &Geometry) -> usize {
617    match g {
618        Geometry::MultiPoint(m) => m.len(),
619        Geometry::MultiLineString(m) => m.0.len(),
620        Geometry::MultiPolygon(m) => m.0.len(),
621        Geometry::GeometryCollection(m) => m.iter().map(part_count).sum(),
622        _ => 1,
623    }
624}

631    #[test]
632    fn multipart_cleaning_keeps_original_positions_and_counts_skipped_rings() {
633        let line = line_string![(x:0.,y:0.),(x:0.,y:0.),(x:1.,y:0.)];
634        let mut polygon = polygon![(x:0.,y:0.),(x:1.,y:0.),(x:1.,y:1.),(x:0.,y:0.)];
635        polygon.interiors_push(LineString::from(vec![(0.5, 0.5); 4]));
636        let geometry = Geometry::GeometryCollection(GeometryCollection(vec![
637            MultiLineString(vec![LineString::empty(), line.clone()]).into(),
638            MultiPolygon(vec![Polygon::empty(), polygon.clone()]).into(),
639            Point::new(4., 5.).into(),
640        ]));
641        let layer = Layer::from_features("parts", [(-12, geometry.clone())]);
642        let parts = layer.parts().unwrap();
643        assert_eq!(parts.skipped, 3);
644        assert_eq!(
645            *parts.lines[0].geometry,
646            line_string![(x:0.,y:0.),(x:1.,y:0.)]
647        );
648        assert_eq!(parts.polygons[0].geometry.exterior(), polygon.exterior());
649        assert!(parts.polygons[0].geometry.interiors().is_empty());
650        for (reference, part) in [
651            (parts.lines[0].reference, 1),
652            (parts.polygons[0].reference, 3),
653            (parts.points[0].reference, 4),
654        ] {
655            assert_eq!(
656                reference.at(LayerId(3)),
657                FeatureRef {
658                    layer: LayerId(3),
659                    fid: -12,
660                    part: Some(part)
661                }
662            );
663        }
664        assert_eq!(layer.features().next().unwrap().1, &geometry);
665    }
