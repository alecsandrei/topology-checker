1//! Incremental validation metadata and sink errors.
2use crate::{Error, RuleReport, RuleSpec, Violation, ViolationKind, Warning};
3use serde::Serialize;
4use std::collections::BTreeMap;
5
6/// One rule's metadata and counts, without retained violation geometry.
7#[derive(Debug, Clone, Serialize)]
8#[non_exhaustive]
9pub struct RuleSummary {
10    /// Registered rule specification.
11    pub spec: RuleSpec,
12    /// Primary layer name.
13    pub layer: String,
14    /// Secondary layer name.
15    pub other: Option<String>,
16    /// Aggregated violation counts.
17    pub counts: BTreeMap<ViolationKind, usize>,
18    /// Total aggregated violations.
19    pub total: usize,
20}
21impl From<&RuleReport> for RuleSummary {
22    fn from(report: &RuleReport) -> Self {
23        Self {
24            spec: report.spec,
25            layer: report.layer.clone(),
26            other: report.other.clone(),
27            counts: report.counts.clone(),
28            total: report.violations.len(),
29        }
30    }
31}
32/// Final metadata available after successful sink exhaustion.
33#[derive(Debug, Clone, Serialize)]
34#[non_exhaustive]
35pub struct ReportSummary {
36    /// Input-coordinate tolerance.
37    pub tolerance: f64,
38    /// Typed input warnings.
39    pub warnings: Vec<Warning>,
40    /// Summaries in rule registration order.
41    pub rules: Vec<RuleSummary>,
42}
43impl ReportSummary {
44    /// Count all aggregated violations.
45    pub fn violation_count(&self) -> usize {
46        self.rules.iter().map(|r| r.total).sum()
47    }
48    /// Whether every rule passed.
49    pub fn is_valid(&self) -> bool {
50        self.violation_count() == 0
51    }
52}
53/// One step of incremental validation, in rule registration order.
54#[derive(Debug)]
55#[non_exhaustive]
56pub enum ValidationEvent<'a> {
57    /// One aggregated violation of a registered rule.
58    Violation(RuleSpec, Violation),
59    /// A rule finished, after all of its violations. Sent for every rule,
60    /// including rules without violations.
61    RuleEnd(&'a RuleSummary),
62}
63/// An operational error or the original error returned by the consumer.
64#[derive(Debug, thiserror::Error)]
65pub enum SinkError<E> {
66    /// Invalid input or geometry preparation failure.
67    #[error(transparent)]
68    Core(#[from] Error),
69    /// Consumer failure, including cancellation. No further output is produced.
70    #[error("validation consumer failed: {0}")]
71    Consumer(E),
72}
73impl From<SinkError<std::convert::Infallible>> for Error {
74    fn from(error: SinkError<std::convert::Infallible>) -> Self {
75        match error {
76            SinkError::Core(error) => error,
77            SinkError::Consumer(never) => match never {},
78        }
79    }
80}
