5macro_rules! registry {
6    ($(($variant:ident, $name:literal, $a:ident, $b:expr, $description:literal, $summary:literal, $arcgis:expr, $qgis:expr)),* $(,)?) => {
7        /// A supported topology rule with its canonical CLI and serialization name.
8        #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
9        #[cfg_attr(feature = "ts", derive(ts_rs::TS))]
10        #[non_exhaustive]
11        pub enum Rule { $( #[doc = $description] #[serde(rename = $name)] $variant, )* }
12        impl Rule {
13            /// All 38 rules in registry order. The original 18 remain the prefix.
14            pub const ALL: &'static [Self] = &[$(Self::$variant),*];
15            /// Return the canonical slash-separated rule name.
16            #[must_use]
17            pub const fn name(self) -> &'static str { match self { $(Self::$variant => $name),* } }
18            /// Return the primary family and optional secondary input family.
19            #[must_use]
20            pub const fn inputs(self) -> (GeometryKind, Option<GeometryKind>) { match self { $(Self::$variant => (GeometryKind::$a, $b)),* } }
21            /// Describe the rule's violation semantics and tolerance behavior.
22            #[must_use]
23            pub const fn description(self) -> &'static str { match self { $(Self::$variant => $description),* } }
24            /// Summarize the rule in plain language for rule catalogs.
25            #[must_use]
26            pub const fn summary(self) -> &'static str { match self { $(Self::$variant => $summary),* } }
27            /// Return the corresponding ArcGIS name, when available.
28            #[must_use]
29            pub const fn arcgis(self) -> Option<&'static str> { match self { $(Self::$variant => $arcgis),* } }
30            /// Return the corresponding QGIS name, when available.
31            #[must_use]
32            pub const fn qgis(self) -> Option<&'static str> { match self { $(Self::$variant => $qgis),* } }
33        }
34    }
35}

77impl Rule {
78    /// Check that a secondary input is supplied exactly when the rule needs one.
79    /// Callers can use this before loading layers; distinct inputs are checked
80    /// at registration, which returns [`Error::SameLayer`](crate::Error::SameLayer).
81    ///
82    /// # Errors
83    /// Returns [`Error::Inputs`](crate::Error::Inputs) with the required layer count.
84    pub fn check_input_count(self, other: bool) -> Result<(), crate::Error> {
85        let binary = self.inputs().1.is_some();
86        if other == binary {
87            Ok(())
88        } else {
89            Err(crate::Error::Inputs {
90                rule: self,
91                expected: if binary { 2 } else { 1 },
92            })
93        }
94    }
95}
96impl fmt::Display for Rule {
97    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
98        f.write_str(self.name())
99    }

101impl FromStr for Rule {
102    type Err = ParseRuleError;
103    /// Parse an exact, case-sensitive canonical rule name.
104    ///
105    /// # Errors
106    /// Returns [`ParseRuleError`] containing the unrecognized input name.
107    fn from_str(value: &str) -> Result<Self, Self::Err> {
108        Self::ALL
109            .iter()
110            .copied()
111            .find(|r| r.name() == value)
112            .ok_or_else(|| ParseRuleError { name: value.into() })
113    }
114}

116/// An unknown canonical rule name encountered during parsing.
117#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
118#[error("unknown rule {name}")]
119pub struct ParseRuleError {
120    name: String,
121}
122impl ParseRuleError {
123    /// Return the original unrecognized name without normalization.
124    #[must_use]
125    pub fn name(&self) -> &str {
126        &self.name
127    }
128}

130/// A registration index scoped to the topology that returned it.
131/// `Report::rules` holds the result for `RuleId` `i` at index `i`.
132#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]

135    /// Return the zero-based rule registration index.
136    #[must_use]

138        self.0
139    }
